# -*- coding: utf-8 -*-
"""
dataset.py - PyTorch Dataset & DataLoader 2.5D cho DuneAI.

Đặc điểm:
  - Đọc trực tiếp từ cache_1mm thông qua mmap (chỉ đọc lát cắt cần dùng).
  - Ghép 3 lát cắt liên tiếp (z-1, z, z+1) tạo đầu vào 2.5D (3, 512, 512).
  - Áp mặt nạ phổi an toàn: Train hợp thêm vùng u giãn nở; Val/Test chỉ dùng phổi (không rò rỉ nhãn).
  - Hỗ trợ chuẩn hóa cửa sổ HU ngay trên CPU hoặc trả về int16 để chạy window_hu_torch trên GPU.

Dùng:
  python dataset.py selftest
  python dataset.py inspect --cache-dir data/cache_1mm
"""
import argparse
import json
import random
import shutil
import sys
import tempfile
from pathlib import Path

import numpy as np
import pandas as pd
import torch
from torch.utils.data import Dataset, DataLoader

THIS_DIR = Path(__file__).resolve().parent
ROOT = THIS_DIR.parent
for p in (str(THIS_DIR), str(ROOT)):
    if p not in sys.path:
        sys.path.insert(0, p)

from configs.config import IMAGE_SIZE
from medical_prep import (
    HU_FLOOR,
    apply_lung_mask,
    clip_and_window_hu,
    load_cached_slices,
    lung_cache_path,
    lung_keep_mask,
    window_hu_torch,
)
from offline_builder import build_dataset_index


class LIDC25DDataset(Dataset):
    def __init__(
        self,
        cache_dir,
        csv_path=None,
        split="train",
        in_channels=3,
        image_size=IMAGE_SIZE,
        apply_lung=True,
        window_on_cpu=True,
        quantize=True,
        augment=False,
    ):
        """
        Args:
            cache_dir: Đường dẫn tới thư mục cache_1mm.
            csv_path: Đường dẫn tới slices.csv (mặc định: <cache_dir>/slices.csv).
            split: 'train', 'val', hoặc 'all'.
            in_channels: Số lát cắt ghép kênh 2.5D (mặc định 3: z-1, z, z+1).
            image_size: Kích thước đầu ra sau pad/crop (mặc định 512).
            apply_lung: Có che vùng ngoài phổi về -1000 HU hay không.
            window_on_cpu: True -> trả về float32 [0, 1]; False -> trả về int16 HU để chạy window_hu_torch trên GPU.
            quantize: Bật/tắt lượng tử hóa bước 25 HU.
            augment: Bật lật/xoay ngẫu nhiên (chỉ có tác dụng khi split='train').
        """
        super().__init__()
        assert in_channels in (1, 3, 5), "in_channels phải là số lẻ (1, 3, hoặc 5)"
        self.cache_dir = Path(cache_dir)
        self.csv_path = Path(csv_path) if csv_path is not None else (self.cache_dir / "slices.csv")
        self.split = split
        self.in_channels = in_channels
        self.half_k = in_channels // 2
        self.image_size = image_size
        self.apply_lung = apply_lung
        self.window_on_cpu = window_on_cpu
        self.quantize = quantize
        self.augment = bool(augment and (split == "train"))

        if not self.csv_path.exists():
            raise FileNotFoundError(
                f"Không tìm thấy {self.csv_path}. Hãy chạy `python offline_builder.py build` trước!"
            )

        df = pd.read_csv(self.csv_path)
        if split is not None and split != "all":
            df = df[df["split"] == split].reset_index(drop=True)
        if len(df) == 0:
            raise ValueError(f"Không có mẫu nào thuộc split='{split}' trong {self.csv_path}")
        self.df = df

    def __len__(self):
        return len(self.df)

    def _neighbor_indices(self, z, num_slices):
        """Lấy chỉ số các lát lân cận, tự động kẹp biên ở lát 0 và lát cuối."""
        return [min(max(0, z + offset), num_slices - 1) for offset in range(-self.half_k, self.half_k + 1)]

    @staticmethod
    def _augment_pair(img_chw, mask_1hw):
        """Biến đổi hình học đồng bộ trên cả khối kênh 2.5D và mask trung tâm."""
        if random.random() < 0.5:
            img_chw = np.flip(img_chw, axis=2)
            mask_1hw = np.flip(mask_1hw, axis=2)
        if random.random() < 0.5:
            img_chw = np.flip(img_chw, axis=1)
            mask_1hw = np.flip(mask_1hw, axis=1)
        k = random.randint(0, 3)
        if k > 0:
            img_chw = np.rot90(img_chw, k=k, axes=(1, 2))
            mask_1hw = np.rot90(mask_1hw, k=k, axes=(1, 2))
        return np.ascontiguousarray(img_chw), np.ascontiguousarray(mask_1hw)

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        pid = str(row["patient_id"])
        z = int(row["z"])
        num_slices = int(row["num_slices"])

        p_dir = self.cache_dir / pid
        z_indices = self._neighbor_indices(z, num_slices)

        use_lung = (
            self.apply_lung
            and bool(row.get("has_lung_mask", 0))
            and lung_cache_path(p_dir).exists()
        )
        loaded = load_cached_slices(p_dir, z_indices, size=self.image_size, with_lung=use_lung)

        if use_lung:
            img_stack, msk_stack, lung_stack = loaded
            # QUAN TRỌNG: Chỉ truyền msk_stack vào lung_keep_mask khi split == 'train'
            tumor_for_keep = msk_stack if self.split == "train" else None
            keep = lung_keep_mask(lung_stack, tumor_mask=tumor_for_keep)
            img_stack = apply_lung_mask(img_stack, keep, fill=HU_FLOOR)
        else:
            img_stack, msk_stack = loaded

        # Nhãn mục tiêu chỉ là lát cắt chính giữa (1, H, W)
        center_mask = (msk_stack[self.half_k : self.half_k + 1] > 0).astype(np.float32)

        if self.window_on_cpu:
            img_out = clip_and_window_hu(img_stack, quantize=self.quantize)
        else:
            img_out = img_stack.astype(np.int16)

        if self.augment:
            img_out, center_mask = self._augment_pair(img_out, center_mask)

        return {
            "image": torch.from_numpy(img_out),
            "mask": torch.from_numpy(center_mask),
            "patient_id": pid,
            "z": z,
        }


def create_dataloaders(
    cache_dir,
    batch_size=8,
    num_workers=2,
    in_channels=3,
    apply_lung=True,
    window_on_cpu=True,
    augment_train=True,
):
    """Hàm tiện ích khởi tạo cặp (train_loader, val_loader) sẵn sàng cho vòng lặp huấn luyện."""
    train_ds = LIDC25DDataset(
        cache_dir=cache_dir,
        split="train",
        in_channels=in_channels,
        apply_lung=apply_lung,
        window_on_cpu=window_on_cpu,
        augment=augment_train,
    )
    val_ds = LIDC25DDataset(
        cache_dir=cache_dir,
        split="val",
        in_channels=in_channels,
        apply_lung=apply_lung,
        window_on_cpu=window_on_cpu,
        augment=False,
    )
    pin = torch.cuda.is_available()
    train_loader = DataLoader(
        train_ds, batch_size=batch_size, shuffle=True, num_workers=num_workers, pin_memory=pin
    )
    val_loader = DataLoader(
        val_ds, batch_size=batch_size, shuffle=False, num_workers=num_workers, pin_memory=pin
    )
    return train_loader, val_loader

# -*- coding: utf-8 -*-
"""
losses.py - Hàm mất mát (Loss Functions) và chỉ số đánh giá (Metrics) cho phân vùng khối u phổi.

Bao gồm:
  - DiceBCELoss: Kết hợp BCEWithLogitsLoss và Batch-level Soft Dice Loss.
  - TverskyFocalLoss: Tùy chọn chuyên trị khối u siêu nhỏ (phạt nặng False Negative).
  - compute_segmentation_metrics: Tính Intersection, Union, Dice, IoU từ raw logits.

Dùng:
  python DuneAI_Remake/src/losses.py selftest
"""
import argparse
import torch
import torch.nn as nn
import torch.nn.functional as F


class DiceBCELoss(nn.Module):
    """
    Hàm mất mát tiêu chuẩn cho phân vùng y khoa:
        Loss = bce_weight * BCEWithLogits + dice_weight * (1 - SoftDice)
    Mặc định bật batch_dice=True để ổn định đạo hàm khi trong batch có lát cắt âm tính (không có u).
    """
    def __init__(
        self,
        bce_weight=0.5,
        dice_weight=0.5,
        pos_weight=None,
        smooth=1.0,
        batch_dice=True,
    ):
        super().__init__()
        self.bce_weight = float(bce_weight)
        self.dice_weight = float(dice_weight)
        self.smooth = float(smooth)
        self.batch_dice = bool(batch_dice)
        if pos_weight is not None:
            self.register_buffer("pos_weight", torch.tensor([float(pos_weight)], dtype=torch.float32))
        else:
            self.pos_weight = None

    def forward(self, logits, targets):
        targets = targets.to(dtype=logits.dtype)
        bce = F.binary_cross_entropy_with_logits(
            logits, targets, pos_weight=self.pos_weight
        )

        probs = torch.sigmoid(logits)
        if self.batch_dice:
            probs_flat = probs.reshape(-1)
            targets_flat = targets.reshape(-1)
            intersection = (probs_flat * targets_flat).sum()
            dice_score = (2.0 * intersection + self.smooth) / (
                probs_flat.sum() + targets_flat.sum() + self.smooth
            )
        else:
            probs_flat = probs.reshape(probs.size(0), -1)
            targets_flat = targets.reshape(targets.size(0), -1)
            intersection = (probs_flat * targets_flat).sum(dim=1)
            dice_score = (2.0 * intersection + self.smooth) / (
                probs_flat.sum(dim=1) + targets_flat.sum(dim=1) + self.smooth
            )
            dice_score = dice_score.mean()

        dice_loss = 1.0 - dice_score
        return self.bce_weight * bce + self.dice_weight * dice_loss


class TverskyFocalLoss(nn.Module):
    """
    Hàm mất mát Tversky-Focal:
      - alpha < beta (mặc định 0.3 / 0.7) giúp phạt lỗi bỏ sót khối u (False Negative) nặng hơn.
      - gamma < 1.0 (hoặc > 1.0 tùy quy ước lũy thừa) tập trung đạo hàm vào các ca khó.
    """
    def __init__(self, alpha=0.3, beta=0.7, gamma=0.75, smooth=1.0):
        super().__init__()
        self.alpha = float(alpha)
        self.beta = float(beta)
        self.gamma = float(gamma)
        self.smooth = float(smooth)

    def forward(self, logits, targets):
        targets = targets.to(dtype=logits.dtype)
        probs = torch.sigmoid(logits).reshape(-1)
        targets_flat = targets.reshape(-1)

        tp = (probs * targets_flat).sum()
        fp = (probs * (1.0 - targets_flat)).sum()
        fn = ((1.0 - probs) * targets_flat).sum()

        tversky = (tp + self.smooth) / (tp + self.alpha * fp + self.beta * fn + self.smooth)
        return torch.pow(1.0 - tversky, self.gamma)


@torch.no_grad()
def compute_segmentation_metrics(logits, targets, threshold=0.5, eps=1e-6):
    """
    Tính toán các đại lượng cơ sở từ raw logits (B, 1, H, W) và nhãn (B, 1, H, W).
    Trả về dict chứa intersection, pred_sum, target_sum để cộng dồn toàn Epoch.
    """
    probs = torch.sigmoid(logits)
    preds = (probs >= threshold).to(torch.float32)
    targets_bin = (targets > 0.5).to(torch.float32)

    preds_f = preds.reshape(-1)
    targets_f = targets_bin.reshape(-1)

    inter = float((preds_f * targets_f).sum().item())
    pred_sum = float(preds_f.sum().item())
    target_sum = float(targets_f.sum().item())
    union = pred_sum + target_sum - inter

    dice = (2.0 * inter + eps) / (pred_sum + target_sum + eps)
    iou = (inter + eps) / (union + eps)

    return {
        "intersection": inter,
        "pred_sum": pred_sum,
        "target_sum": target_sum,
        "dice": dice,
        "iou": iou,
    }


def _cmd_selftest(_):
    print("Đang chạy kiểm thử tự động losses.py...")
    targets = torch.zeros((4, 1, 64, 64), dtype=torch.float32)
    targets[0:3, 0, 16:32, 16:32] = 1.0  # 3 lát có u, 1 lát âm tính

    # 1. Dự đoán hoàn hảo (logits = +10 ở vùng u, -10 ở nền)
    logits_good = torch.where(targets > 0.5, torch.tensor(10.0), torch.tensor(-10.0))
    logits_good.requires_grad_(True)

    criterion = DiceBCELoss(bce_weight=0.5, dice_weight=0.5)
    loss_good = criterion(logits_good, targets)
    loss_good.backward()

    m_good = compute_segmentation_metrics(logits_good, targets)
    assert loss_good.item() < 0.01, f"Loss dự đoán đúng phải ~0, nhận được: {loss_good.item()}"
    assert m_good["dice"] > 0.999 and m_good["iou"] > 0.999
    assert logits_good.grad is not None and not torch.isnan(logits_good.grad).any()

    # 2. Dự đoán sai hoàn toàn (ngược với nhãn)
    logits_bad = torch.where(targets > 0.5, torch.tensor(-10.0), torch.tensor(10.0))
    loss_bad = criterion(logits_bad, targets)
    m_bad = compute_segmentation_metrics(logits_bad, targets)
    assert loss_bad.item() > 5.0 and m_bad["dice"] < 0.001

    # 3. Kiểm thử TverskyFocalLoss
    tf_loss = TverskyFocalLoss()(logits_good.detach(), targets)
    assert tf_loss.item() < 0.01

    print(
        f"✅ selftest losses.py: OK! "
        f"(Good Loss={loss_good.item():.5f}, Dice={m_good['dice']:.4f} | "
        f"Bad Loss={loss_bad.item():.2f}, Dice={m_bad['dice']:.4f})"
    )

    sub = ap.add_subparsers(dest="cmd", required=False)
    s = sub.add_parser("selftest", help="Kiểm thử hàm Loss và Metrics")
    s.set_defaults(fn=_cmd_selftest)
    args = ap.parse_args()
    _cmd_selftest(args)

def load_nrrd(file_path):
    """
    Đọc file NRRD bằng SimpleITK.
    Trả về:
        - volume: mảng numpy 3D (Z, H, W)
        - spacing: tuple (spacing_x, spacing_y, spacing_z) theo mm  (thứ tự của SimpleITK)

def window_bounds(ww=WINDOW_WIDTH, wl=WINDOW_LEVEL):
    """Giống _apply_window của tác giả: cận dưới bị kẹp ở -1000. Mặc định (1500, -600) -> [-1000, 150]."""
    assert 0 < ww < 2000 and -1000 < wl < 200, "WW/WL ngoài miền hợp lệ của tác giả"
    low = max(wl - ww / 2.0, HU_FLOOR)
    high = wl + ww / 2.0
    return low, high


def clip_and_window_hu(volume, ww=WINDOW_WIDTH, wl=WINDOW_LEVEL, step=RESAMPLING_STEP, quantize=True):
    """
    Phiên bản numpy (CPU). Đầu vào là HU THÔ (nên là ảnh đã resize và pad -1000).
    1) cắt đáy -1000  2) kẹp cửa sổ  3) lượng tử hóa theo `step`  4) chuẩn hóa về [0, 1].
    Không sửa mảng đầu vào. Đầu ra float32; không khí = 0.
    """
    low, high = window_bounds(ww, wl)
    vol = np.clip(volume.astype(np.float32), low, high)          # gộp bước cắt đáy và cửa sổ
    if quantize:                                                  # bin = floor((HU - low) / step)
        vol = np.floor((vol - low) / step) * step + low
    return (vol - low) / (high - low)


def window_hu_torch(x, ww=WINDOW_WIDTH, wl=WINDOW_LEVEL, step=RESAMPLING_STEP, quantize=True, out_dtype=None):
    """
    Phiên bản GPU: cùng công thức với clip_and_window_hu nhưng chạy trên tensor (cuda).
    x: tensor int16/float bất kỳ shape (HU). out_dtype: ví dụ torch.float16 để tiết kiệm VRAM.
    """
    assert torch is not None, "Cần cài PyTorch để dùng window_hu_torch"
    low, high = window_bounds(ww, wl)
    x = torch.clamp(x.to(torch.float32), low, high)
    if quantize:
        x = torch.floor((x - low) / step) * step + low
    x = (x - low) / (high - low)
    return x if out_dtype is None else x.to(out_dtype)

def resize_to_1mm_3d(volume, spacing, is_mask=False, interp=IMAGE_INTERP):
    """
    Đưa mặt phẳng trục về 1 mm/pixel (CHƯA pad). Trục z giữ nguyên như tác giả.
    Mask luôn dùng nearest để không tạo giá trị trung gian.
    """
    sx, sy = float(spacing[0]), float(spacing[1])
    z, h, w = volume.shape
    new_h, new_w = max(1, int(round(h * sy))), max(1, int(round(w * sx)))

    if is_mask:
        src = (volume > 0).astype(np.uint8)
        flag = cv2.INTER_NEAREST
    else:
        src = np.ascontiguousarray(volume)
        flag = _CV2_INTERP[interp]

    out = np.empty((z, new_h, new_w), dtype=src.dtype)
    for i in range(z):
        out[i] = cv2.resize(src[i], (new_w, new_h), interpolation=flag)
    return out


def _axis_window(n, size):
    """Trả về (bắt đầu nguồn, bắt đầu đích, độ dài) khi crop (n>=size) hoặc pad (n<size) vào giữa."""
    if n >= size:
        return (n - size) // 2, 0, size
    return 0, (size - n) // 2, n


def pad_or_crop_xy(x, size=IMAGE_SIZE, pad_value=HU_FLOOR):
    """
    Crop hoặc pad hai chiều cuối về (size, size), đặt giữa. Dùng được cho numpy (kể cả memmap) và torch.
    Ảnh HU: pad_value = -1000. Mask: pad_value = 0.
    """
    h, w = x.shape[-2:]
    sy, dy, ly = _axis_window(h, size)
    sx, dx, lx = _axis_window(w, size)
    shape = (*x.shape[:-2], size, size)
    if torch is not None and isinstance(x, torch.Tensor):
        out = torch.full(shape, pad_value, dtype=x.dtype, device=x.device)
    else:
        out = np.full(shape, pad_value, dtype=x.dtype)
    out[..., dy:dy + ly, dx:dx + lx] = x[..., sy:sy + ly, sx:sx + lx]
    return out


def resize_and_pad_3d(volume, spacing, target_size=IMAGE_SIZE, is_mask=False, pad_value=None, interp=IMAGE_INTERP):
    """
    Một pixel ảnh = 1 mm thực tế, rồi crop/pad về target_size x target_size.
    pad_value mặc định: -1000 (HU, không khí) cho ảnh; 0 cho mask.
    LƯU Ý: truyền HU THÔ (chưa cửa sổ). Nếu truyền ảnh đã chuẩn hóa thì phải tự đặt pad_value = 0.
    """
    if pad_value is None:
        pad_value = 0 if is_mask else HU_FLOOR
    small = resize_to_1mm_3d(volume, spacing, is_mask=is_mask, interp=interp)
    return pad_or_crop_xy(small, target_size, pad_value)

    z_dist = np.sum(image, axis=(1, 2))
    zi = np.where(z_dist != 0)
    return int((np.max(zi) - np.min(zi)) / 2), int(np.min(zi)), int(np.max(zi))


def _largest_labeled_volumes(im, bg=-1, ratio=0.6):
    """
    Giữ nguyên hành vi của tác giả: trả về nhãn lớn nhất; chỉ khi có ĐÚNG MỘT nhãn có thể tích
    trong (ratio, 1) so với nhãn lớn nhất thì trả thêm nhãn đó (để gộp phổi trái/phải bị tách).
    """
    vals, counts = np.unique(im, return_counts=True)
    counts = counts[vals != bg]
    vals = vals[vals != bg]
    if len(counts) == 0:
        return None
    counts = counts / np.max(counts)
    cand = np.where((counts > ratio) & (counts < 1))[0]
    if len(cand) == 1:
        smax = np.where(counts == counts[cand[0]])[0][0]
        return vals[[np.argmax(counts), smax]]
    return [vals[np.argmax(counts)]]


def _create_flipped_mask(image, i_orig, kernel):
    """
    method=0 của tác giả: tìm tâm phổi theo phân bố xương (> 100 HU) rồi lật đối xứng mask qua tâm
    để mask phổi bao cả khối đặc dính thành ngực/trung thất. Trả về (mask, min_bound, max_bound).
    """
    _, min_bound, max_bound = _find_bounds_z_direction(image)
    fl_mask = np.zeros(image.shape, np.uint8)

    lc_buff = []
    for i in range(min_bound, max_bound + 1):
        rib_view = 1 * (i_orig[i, ...] > 100)
        distr = np.sum(rib_view, axis=0)
        thresh_l1 = np.max(distr)
        if thresh_l1 == 0:                    # (thêm) lát không có xương: bỏ qua
            continue
        idx = np.where(distr > (thresh_l1 * 0.15))[0]
        lc_buff.append((np.min(idx) + np.max(idx)) / 2)
    lungs_center = int(np.mean(lc_buff)) if lc_buff else image.shape[2] // 2
    half = int(image.shape[2] / 2)
    W = image.shape[2]

    for i in range(min_bound, max_bound + 1):
        ax = image[i, ...]
        if lungs_center > half:
            part1 = ax[:, 2 * lungs_center - W:lungs_center]
            part2 = np.hstack((ax[:, lungs_center:], np.zeros((image.shape[1], 2 * lungs_center - W))))
            part2 = cv2.dilate(np.array(part2, np.uint8), kernel, iterations=1)
            part1 = cv2.dilate(np.array(part1, np.uint8), kernel, iterations=1)
            fl_mask[i, ...] = np.hstack((np.flip(part2, axis=1), np.flip(part1, axis=1)))
            fl_mask[i, ...] = np.array(fl_mask[i, ...], np.uint8) | np.array(ax, np.uint8)
        elif lungs_center < half:
            part1 = np.hstack((np.zeros((image.shape[1], W - 2 * lungs_center)), ax[:, :lungs_center]))
            part2 = ax[:, lungs_center:W - (W - 2 * lungs_center)]
            part2 = cv2.dilate(np.array(part2, np.uint8), kernel, iterations=1)
            part1 = cv2.dilate(np.array(part1, np.uint8), kernel, iterations=1)
            fl_mask[i, ...] = np.hstack((np.flip(part2, axis=1), np.flip(part1, axis=1)))
            fl_mask[i, ...] = np.array(fl_mask[i, ...], np.uint8) | np.array(ax, np.uint8)
        else:
            fl = np.flip(ax, axis=1)
            fl_mask[i, ...] = np.array(ax, np.uint8) | np.array(fl, np.uint8)
    return fl_mask, min_bound, max_bound


def extract_lungs_mask(volume, return_bounds=False):
    """
    Chuyển thể get_seg_lungs của tác giả. Đầu vào: HU thô (Z, H, W) ở lưới gốc.
    Các bước: ngưỡng < -320 HU -> clear_border (bỏ không khí ngoài, bàn chụp) -> giữ 3 vùng lớn nhất
    -> roberts + binary_fill_holes (lấp mạch máu) -> bỏ khí quản (vùng nhỏ hơn 0.0069*sq/512^2)
    -> bỏ bàn/túi khí (tâm khối nằm dưới 75% chiều cao) -> giữ khối liên thông 3D lớn nhất
    -> lật đối xứng qua tâm cột sống -> morphology -> kéo dãn +-2 lát theo z.
    Trả về mask uint8 (Z, H, W) gồm 0/1, hoặc None nếu không tìm thấy phổi.
    Chạy chậm (vòng lặp từng lát) nên chỉ gọi trong build, không gọi trong vòng huấn luyện.
    """
    label, regionprops, clear_border, roberts, ndi = _import_lung_deps()
    im = np.asarray(volume)
    kernel = _get_kernel()
    sq = im.shape[1] * im.shape[2]
    seg = np.zeros(im.shape, np.uint8)

    # --- ngưỡng, bỏ vùng chạm biên, giữ 3 vùng lớn nhất, lấp lỗ ---
    for nm, ax_slice in enumerate(im < LUNG_THRESHOLD_HU):
        lab = label(clear_border(ax_slice))
        areas = sorted(r.area for r in regionprops(lab))
        if len(areas) > 3:
            for r in regionprops(lab):
                if r.area < areas[-3]:
                    lab[lab == r.label] = 0
        seg[nm] = ndi.binary_fill_holes(roberts(lab > 0))

    # --- bỏ khí quản: vùng nhỏ hơn TRACHEA_COEFF * (sq / 512^2) diện tích ảnh ---
    trachea_coeff = TRACHEA_COEFF * (sq / (512 ** 2))
    for i, ax_slice in enumerate(seg):
        lab = label(ax_slice, connectivity=1, background=0)
        vals, counts = np.unique(lab, return_counts=True)
        counts, vals = counts[vals != 0], vals[vals != 0]
        small = counts / sq < trachea_coeff
        if small.any():
            for v in vals[small]:
                lab[lab == v] = 0
            seg[i] = lab != 0

    # --- bỏ bàn chụp và túi khí: tâm khối ở dưới 75% chiều cao ảnh ---
    lab3 = label(seg, background=0)
    ids = np.unique(lab3)
    ids = ids[ids != 0]
    if len(ids):
        centers = ndi.center_of_mass(np.ones_like(lab3), lab3, ids)
        for i, c in zip(ids, centers):
            if c[1] > seg.shape[1] * 0.75:
                seg[lab3 == i] = 0

    # --- giữ khối liên thông 3D lớn nhất (gộp phổi trái/phải nếu tách đôi) ---
    lab3 = label(seg, background=0)
    l_max = _largest_labeled_volumes(lab3, bg=0)
    if l_max is not None:
        if len(l_max) > 1:
            lab3[lab3 == l_max[1]] = l_max[0]
        seg[lab3 != l_max[0]] = 0
    if seg.sum() == 0:
        return None

    # --- lật đối xứng qua tâm cột sống ---
    seg, min_bound, max_bound = _create_flipped_mask(seg, im, kernel)

    # --- morphology: lấp khoảng trống nhỏ trong mask ---
    for i, ax_slice in enumerate(seg):
        bi = np.array(ax_slice, np.uint8)
        opening = cv2.morphologyEx(1 - bi, cv2.MORPH_OPEN, kernel, iterations=7)
        dilat = cv2.dilate(1 - opening, kernel, iterations=2)
        seg[i, ...] = cv2.erode(dilat, kernel, iterations=1)

    # --- kéo dãn mask +-2 lát theo z ---
    if (min_bound > 2) and (max_bound < seg.shape[0] - 2):
        for i in range(1, 3):
            seg[min_bound - i, ...] = seg[min_bound, ...]
            seg[max_bound + i, ...] = seg[max_bound, ...]
        min_bound -= 2
        max_bound += 2

    seg = (seg > 0).astype(np.uint8)
    return (seg, min_bound, max_bound) if return_bounds else seg


def lung_keep_mask(lung_mask, tumor_mask=None, tumor_dilate_iter=3):
    """
    Vùng được giữ lại = phổi, hợp với mask khối u/nốt đã dilate (kernel 11x11, 3 lần) - như generator dòng 132-134.
    CHỈ truyền tumor_mask lúc HUẤN LUYỆN. Lúc đánh giá/dự đoán mà truyền mask thật vào thì ảnh đầu vào đã
    chứa thông tin nhãn (rò rỉ).
    """
    keep = (lung_mask > 0).astype(np.uint8)
    if tumor_mask is not None:
        kernel = _get_kernel()
        for k in range(len(keep)):
            if tumor_mask[k].any():
                keep[k] |= cv2.dilate((tumor_mask[k] > 0).astype(np.uint8), kernel, iterations=tumor_dilate_iter)
    return keep


def apply_lung_mask(volume_hu, keep, fill=HU_FLOOR):
    """Đặt mọi voxel ngoài vùng giữ lại về -1000 HU (apply_mask của tác giả). Không sửa mảng đầu vào."""
    out = np.array(volume_hu, copy=True)
    out[np.asarray(keep) == 0] = fill
    return out

CACHE_VERSION = 2


def cache_paths(out_dir):
    out_dir = Path(out_dir)
    return out_dir / "image_1mm.npy", out_dir / "mask_1mm.npy", out_dir / "meta.json"


def lung_cache_path(out_dir):
    return Path(out_dir) / "lung_1mm.npy"


def _save_npy(path, arr):
    tmp = Path(str(path) + ".tmp")
    with open(tmp, "wb") as f:
        np.save(f, arr)
    os.replace(tmp, path)


def build_patient_cache(patient_dir, out_dir, interp=IMAGE_INTERP, overwrite=False, extract_lungs=False):
    """
    Bước 1-3 cho một bệnh nhân: đọc image.nrrd (+ mask.nrrd), cắt đáy -1000, (tùy chọn) tách phổi, resize 1 mm.
    Ảnh lưu int16 HU, mask uint8. Mask phổi lưu riêng (lung_1mm.npy), KHÔNG áp vào ảnh trong cache.
    Pad về IMAGE_SIZE làm lúc nạp (rất rẻ) để cache nhỏ hơn.
    """
    patient_dir, out_dir = Path(patient_dir), Path(out_dir)
    img_p, msk_p, meta_p = cache_paths(out_dir)
    lung_p = lung_cache_path(out_dir)
    done = img_p.exists() and msk_p.exists() and meta_p.exists() and (lung_p.exists() or not extract_lungs)
    if not overwrite and done:
        return "skip"

    image_nrrd, mask_nrrd = patient_dir / "image.nrrd", patient_dir / "mask.nrrd"
    if not image_nrrd.exists():
        return "missing_image"

    vol, spacing = load_nrrd(image_nrrd)
    vol = np.maximum(vol, HU_FLOOR).astype(np.int16)             # bước 1: cắt đáy -1000 HU

    if mask_nrrd.exists():
        msk, _ = load_nrrd(mask_nrrd)
        if msk.shape != vol.shape:
            raise ValueError(f"mask {msk.shape} khác image {vol.shape}")
    else:
        msk = np.zeros_like(vol, dtype=np.uint8)

    lung1, lung_meta = None, {}
    if extract_lungs:                                            # bước 2: tách phổi trên lưới gốc
        lung = extract_lungs_mask(vol)
        if lung is None:
            lung_meta = {"lung_ok": False}
            lung1 = np.zeros((vol.shape[0], ) + resize_to_1mm_3d(vol[:1], spacing).shape[1:], np.uint8)
        else:
            n_nod = int((msk > 0).sum())
            lung_meta = {
                "lung_ok": True,
                "lung_voxels_orig": int(lung.sum()),
                # tỉ lệ voxel nốt nằm trong mask phổi: thấp nghĩa là che ảnh bằng mask phổi sẽ cắt mất nốt
                "mask_in_lung_fraction": (float(((msk > 0) & (lung > 0)).sum() / n_nod) if n_nod else None),
            }
            lung1 = resize_to_1mm_3d(lung, spacing, is_mask=True)

    img1 = resize_to_1mm_3d(vol, spacing, is_mask=False, interp=interp)      # bước 3
    msk1 = resize_to_1mm_3d(msk, spacing, is_mask=True)

    sx, sy, sz = (float(s) for s in spacing)
    meta = {
        "cache_version": CACHE_VERSION,
        "interp": interp,
        "orig_shape_zyx": list(vol.shape),
        "orig_spacing_xyz": [sx, sy, sz],
        "shape_1mm_zyx": list(img1.shape),
        "mask_voxels_orig": int((msk > 0).sum()),
        "mask_voxels_1mm": int(msk1.sum()),
        "mask_volume_mm3_orig": float((msk > 0).sum() * sx * sy * sz),
        "mask_volume_mm3_1mm": float(msk1.sum() * 1.0 * 1.0 * sz),
        **lung_meta,
    }
    out_dir.mkdir(parents=True, exist_ok=True)
    _save_npy(img_p, img1)
    _save_npy(msk_p, msk1)
    if lung1 is not None:
        _save_npy(lung_p, lung1)
    with open(meta_p, "w", encoding="utf-8") as f:
        json.dump(meta, f, ensure_ascii=False, indent=2)
    return "ok"


def load_cached(cache_dir, mmap=True):
    """Trả về (image_1mm, mask_1mm, meta). Với mmap=True chỉ đọc phần dữ liệu thật sự được truy cập."""
    img_p, msk_p, meta_p = cache_paths(cache_dir)
    mode = "r" if mmap else None
    img = np.load(img_p, mmap_mode=mode)
    msk = np.load(msk_p, mmap_mode=mode)
    with open(meta_p, "r", encoding="utf-8") as f:
        meta = json.load(f)
    return img, msk, meta


def load_cached_slices(cache_dir, z_indices, size=IMAGE_SIZE, with_lung=False):
    """
    Nạp các lát z_indices đã pad về (K, size, size): ảnh int16 HU (pad -1000), mask uint8 (pad 0),
    và (nếu with_lung) mask phổi uint8. Đầu ra là HU thô -> áp lung_keep_mask/apply_lung_mask nếu cần,
    chuyển lên GPU rồi gọi window_hu_torch.
    """
    img, msk, _ = load_cached(cache_dir, mmap=True)
    z = np.asarray(z_indices)
    out = [pad_or_crop_xy(np.asarray(img[z]), size, HU_FLOOR),
           pad_or_crop_xy(np.asarray(msk[z]), size, 0)]
    if with_lung:
        lung = np.load(lung_cache_path(cache_dir), mmap_mode="r")
        out.append(pad_or_crop_xy(np.asarray(lung[z]), size, 0))
    return tuple(out)

def _worker(args):
    pid, nrrd_dir, cache_dir, interp, overwrite, extract_lungs = args
    cv2.setNumThreads(1)                       # tránh tranh chấp luồng khi chạy nhiều tiến trình
    try:
        return pid, build_patient_cache(Path(nrrd_dir) / pid, Path(cache_dir) / pid, interp, overwrite, extract_lungs)
    except Exception as e:                     # một ca lỗi không làm dừng cả lượt chạy
        return pid, f"error: {e}"


def _cmd_build(a):
    nrrd_dir, cache_dir = Path(a.nrrd_dir), Path(a.cache_dir)
    pids = sorted(p.name for p in nrrd_dir.iterdir() if p.is_dir())
    if a.limit:
        pids = pids[:a.limit]
    print(f"{len(pids)} bệnh nhân -> {cache_dir}" + (" (có tách phổi, chậm hơn nhiều)" if a.extract_lungs else ""))
    jobs = [(p, str(nrrd_dir), str(cache_dir), a.interp, a.overwrite, a.extract_lungs) for p in pids]

    try:
        from tqdm import tqdm
    except ImportError:
        tqdm = lambda x, **k: x                # noqa: E731

    results = []
    if a.workers > 1:
        with ProcessPoolExecutor(max_workers=a.workers) as ex:
            for r in tqdm(ex.map(_worker, jobs), total=len(jobs)):
                results.append(r)
    else:
        for j in tqdm(jobs):
            results.append(_worker(j))

    bad = [(p, s) for p, s in results if s not in ("ok", "skip")]
    print(f"Xong: {sum(s == 'ok' for _, s in results)} mới, {sum(s == 'skip' for _, s in results)} bỏ qua, {len(bad)} lỗi")
    for p, s in bad:
        print("  ", p, s)


def _phantom_ct(z=14):
    """CT tổng hợp: thân người, hai phổi, khí quản, cột sống (xương), bàn chụp, không khí ngoài khung."""
    yy, xx = np.mgrid[:512, :512]
    vol = np.full((z, 512, 512), -1000, np.int16)
    vol[:, ((yy - 256) / 170.0) ** 2 + ((xx - 256) / 225.0) ** 2 <= 1] = 40          # thân
    lung_l = ((yy - 250) / 110.0) ** 2 + ((xx - 140) / 60.0) ** 2 <= 1
    lung_r = ((yy - 250) / 110.0) ** 2 + ((xx - 372) / 60.0) ** 2 <= 1
    vol[:, lung_l | lung_r] = -800                                                    # phổi
    vol[:, (yy - 230) ** 2 + (xx - 256) ** 2 <= 100] = -950                           # khí quản
    vol[:, (yy - 340) ** 2 + (xx - 256) ** 2 <= 18 ** 2] = 300                        # cột sống
    vol[:, 432:438, 110:400] = 60                                                     # bàn chụp
    return vol, (lung_l | lung_r)

    big = resize_and_pad_3d(vol, (1.5, 1.5, 2.5))                    # 768 mm > 512 -> crop
    assert big.shape == (6, 512, 512)
    print("resize/pad/cửa sổ: OK")

    if torch is not None:
        t = window_hu_torch(torch.from_numpy(r)).numpy()

    else:
        print("(bỏ qua so sánh torch: chưa cài PyTorch)")

        _import_lung_deps()
    except ImportError:

    else:

        union = (lung[mid] > 0) | truth
        iou = inter.sum() / union.sum()
        assert iou > 0.8, f"IoU phổi quá thấp: {iou:.3f}"
        assert lung[mid, 230, 256] == 0, "khí quản phải bị loại"
        assert lung[mid, 5, 5] == 0 and lung[mid, 435, 250] == 0, "không khí ngoài và bàn chụp phải bị loại"
        print(f"tách phổi trên phantom: OK (IoU = {iou:.3f})")
    print("selftest OK")

# -*- coding: utf-8 -*-
"""
custom_backbone.py - Bộ mã hóa (Encoder) tùy chỉnh tích hợp MambaVision cho DuneAI.

Đặc điểm thiết kế:
  1. Lazy Import (Nạp trễ): Chỉ nạp thư viện `transformers` / `mambavision` khi thực sự
     bật MambaVisionEncoder. Khi chạy Baseline DuneUNet (`backbone="none"`), hệ thống
     hoàn toàn không yêu cầu cài đặt các thư viện ngoài này.
  2. Tự động nhận diện số kênh (Dynamic Channel Probing): Tự động dò kích thước 4 tầng
     đặc trưng (Stage 1..4) của mọi biến thể MambaVision (T, T2, S, B, L).
  3. Cơ chế Đóng / Mở khóa trọng số (Freeze / Unfreeze Backbone):
     - `freeze()`: Khóa toàn bộ tham số backbone (`requires_grad=False`) và khóa các lớp
       BatchNorm/LayerNorm ở chế độ `.eval()` để bảo toàn tri thức Pre-trained.
     - `unfreeze()`: Mở khóa toàn bộ tham số (`requires_grad=True`) để Fine-tune toàn mạng.
  4. Hỗ trợ Mock Backbone thuần PyTorch (`use_mock=True`) để chạy `selftest` offline
     mà không cần internet hay nhân CUDA `mamba-ssm`.

Dùng:
  python DuneAI_Remake/src/models/custom_backbone.py selftest
"""
import argparse
import torch
import torch.nn as nn


# Bảng ánh xạ tên gọi ngắn gọn sang kho lưu trữ Hugging Face chính thức của NVIDIA
MAMBAVISION_REGISTRY = {
    "mambavision_t": "nvidia/MambaVision-T-1K",
    "mambavision_t2": "nvidia/MambaVision-T2-1K",
    "mambavision_s": "nvidia/MambaVision-S-1K",
    "mambavision_b": "nvidia/MambaVision-B-1K",
    "mambavision_l": "nvidia/MambaVision-L-1K",
    "mambavision_l2": "nvidia/MambaVision-L2-1K",
}


# Kênh cơ sở `dim` của từng biến thể (theo mã nguồn NVlabs/MambaVision); 4 tầng có số kênh dim * 2^i.
# MambaVision cần nhân CUDA `mamba-ssm` nên KHÔNG chạy được trên CPU -> không thể "dò" bằng forward ở đó.
MAMBAVISION_BASE_DIM = {
    "mambavision_t": 80,
    "mambavision_t2": 80,
    "mambavision_s": 96,
    "mambavision_b": 128,
    "mambavision_l": 196,
    "mambavision_l2": 196,
}


def is_mambavision_backbone(backbone_name) -> bool:
    """Kiểm tra xem tên backbone có kích hoạt MambaVision hay không (False = dùng U-Net thuần CNN)."""
    if backbone_name is None:
        return False
    name = str(backbone_name).strip().lower()
    return name not in ("", "none", "null", "false", "unet", "cnn", "baseline")


class _MockMambaVisionBackbone(nn.Module):
    """
    Mô hình giả lập (Mock) mô phỏng chính xác cấu trúc 4 tầng phân cấp của MambaVision-T:
      - Stage 1: (B,  80, H/4,  W/4)
      - Stage 2: (B, 160, H/8,  W/8)
      - Stage 3: (B, 320, H/16, W/16)
      - Stage 4: (B, 640, H/32, W/32)
    Phục vụ kiểm thử tự động (`selftest`) trên mọi môi trường CPU/GPU offline.
    """
    def __init__(self, dims=(80, 160, 320, 640)):
        super().__init__()
        self.dims = list(dims)
        self.stage1 = nn.Sequential(
            nn.Conv2d(3, dims[0], kernel_size=4, stride=4, bias=False),
            nn.BatchNorm2d(dims[0]),
            nn.GELU(),
        )
        self.stage2 = nn.Sequential(
            nn.Conv2d(dims[0], dims[1], kernel_size=2, stride=2, bias=False),
            nn.BatchNorm2d(dims[1]),
            nn.GELU(),
        )
        self.stage3 = nn.Sequential(
            nn.Conv2d(dims[1], dims[2], kernel_size=2, stride=2, bias=False),
            nn.BatchNorm2d(dims[2]),
            nn.GELU(),
        )
        self.stage4 = nn.Sequential(
            nn.Conv2d(dims[2], dims[3], kernel_size=2, stride=2, bias=False),
            nn.BatchNorm2d(dims[3]),
            nn.GELU(),
        )

    def forward(self, x):
        f1 = self.stage1(x)
        f2 = self.stage2(f1)
        f3 = self.stage3(f2)
        f4 = self.stage4(f3)
        out_avg_pool = f4.mean(dim=(-2, -1))
        return out_avg_pool, [f1, f2, f3, f4]


class MambaVisionEncoder(nn.Module):
    """
    Bộ chuyển đổi (Adapter) trích xuất 4 tầng đặc trưng đa tỷ lệ từ MambaVision (NVIDIA)
    để truyền qua các nhánh Skip Connections của mạng U-Net.
    """
    def __init__(
        self,
        model_name="mambavision_t",
        in_channels=3,
        pretrained=True,
        freeze_backbone=False,
        normalize_imagenet=True,
        use_mock=False,
    ):
        super().__init__()
        self.model_name = str(model_name)
        self.in_channels = int(in_channels)
        self.pretrained = bool(pretrained)
        self.normalize_imagenet = bool(normalize_imagenet)
        self.use_mock = bool(use_mock)
        self._is_frozen = False

        # Nếu số lát cắt đầu vào khác 3 (VD: 1 hoặc 5 kênh), dùng Conv 1x1 ánh xạ về 3 kênh chuẩn
        if self.in_channels != 3:
            self.input_adapter = nn.Conv2d(self.in_channels, 3, kernel_size=1, bias=False)
        else:
            self.input_adapter = nn.Identity()

        # Chuẩn hóa ảnh CT [0, 1] theo phân phối ImageNet mà MambaVision đã học
        self.register_buffer("img_mean", torch.tensor([0.485, 0.456, 0.406]).view(1, 3, 1, 1))
        self.register_buffer("img_std", torch.tensor([0.229, 0.224, 0.225]).view(1, 3, 1, 1))

        key = self.model_name.strip().lower()
        self.hf_repo = MAMBAVISION_REGISTRY.get(key, self.model_name)

        self._key = key
        self._is_mock = bool(self.use_mock or key == "mambavision_mock")
        if self._is_mock:
            self.backbone = _MockMambaVisionBackbone()
        else:
            self.backbone = self._load_real_mambavision(self.hf_repo, pretrained=self.pretrained)

        # Tự động dò số kênh của 4 stages [C1, C2, C3, C4]
        self.out_channels = self._probe_out_channels()
        self.stem_stride = 4

        if freeze_backbone:
            self.freeze()

    @staticmethod
    def _load_real_mambavision(hf_repo, pretrained=True):
        try:
            from transformers import AutoConfig, AutoModel
        except ImportError as e:
            raise ImportError(
                "Chưa cài đặt thư viện `transformers` để nạp MambaVision. Bản thật còn cần timm, einops và "
                "`mamba-ssm` (nhân CUDA, cần Linux + GPU NVIDIA): "
                "`pip install transformers timm einops mamba-ssm`. Muốn thử trên CPU hãy dùng use_mock=True."
            ) from e

        if not torch.cuda.is_available():
            print("[MambaVisionEncoder] ⚠️ Không có CUDA: MambaVision thật dùng nhân CUDA `selective_scan_fn` "
                  "nên sẽ KHÔNG chạy được forward trên CPU.")

        if pretrained:
            print(f"[MambaVisionEncoder] Đang tải Pre-trained Backbone từ: {hf_repo} ...")
            return AutoModel.from_pretrained(hf_repo, trust_remote_code=True)
        else:
            print(f"[MambaVisionEncoder] Khởi tạo kiến trúc MambaVision từ đầu (scratch): {hf_repo} ...")
            cfg = AutoConfig.from_pretrained(hf_repo, trust_remote_code=True)
            return AutoModel.from_config(cfg, trust_remote_code=True)

    def _probe_out_channels(self):
        if not self._is_mock and self._key in MAMBAVISION_BASE_DIM:
            d = MAMBAVISION_BASE_DIM[self._key]
            return [d * 2 ** i for i in range(4)]

        was_training = self.backbone.training
        self.backbone.eval()
        with torch.no_grad():
            param = next(self.backbone.parameters(), None)
            device = param.device if param is not None else torch.device("cpu")
            if not self._is_mock and device.type != "cuda":
                raise RuntimeError(
                    f"Không biết số kênh của '{self.model_name}' và không thể dò bằng forward ngoài CUDA. "
                    f"Hãy dùng một tên trong {sorted(MAMBAVISION_BASE_DIM)} hoặc tạo encoder trên GPU."
                )
            dummy = torch.zeros(1, 3, 128, 128, device=device)
            out = self.backbone(dummy)
            features = self._extract_feature_list(out)
            if len(features) != 4:
                raise RuntimeError(
                    f"MambaVisionEncoder yêu cầu đúng 4 tầng đặc trưng, nhưng nhận được {len(features)}"
                )
            channels = [int(f.shape[1]) for f in features]
        self.backbone.train(was_training)
        return channels

    @staticmethod
    def _extract_feature_list(backbone_output):
        """Trích xuất danh sách 4 feature maps từ đầu ra của MambaVision."""
        if isinstance(backbone_output, (tuple, list)) and len(backbone_output) == 2:
            _, features = backbone_output
            if isinstance(features, (tuple, list)):
                return list(features)
        if isinstance(backbone_output, dict) and "features" in backbone_output:
            return list(backbone_output["features"])
        if hasattr(backbone_output, "features"):
            return list(backbone_output.features)
        hs = getattr(backbone_output, "hidden_states", None)
        if hs is None and isinstance(backbone_output, dict):
            hs = backbone_output.get("hidden_states")
        if hs is not None:
            return list(hs)
        raise TypeError(
            f"Định dạng đầu ra của MambaVision không hợp lệ: {type(backbone_output)}. "
            f"Cần (pooled, [4 feature maps]) hoặc đối tượng có `features`/`hidden_states`."
        )

    # =========================================================================
    # CƠ CHẾ ĐÓNG / MỞ KHÓA TRỌNG SỐ (FREEZE / UNFREEZE BACKBONE)
    # =========================================================================
    def freeze(self):
        """
        Đóng băng trọng số lõi MambaVision (`requires_grad = False`) và khóa BatchNorm/LayerNorm
        ở chế độ `.eval()`. `input_adapter` (nếu in_channels != 3) vẫn được học; khi đó forward giữ đồ thị
        autograd xuyên qua backbone đóng băng nên tốn thêm bộ nhớ kích hoạt.
        """
        self._is_frozen = True
        for param in self.backbone.parameters():
            param.requires_grad = False
        self.backbone.eval()
        return self

    def unfreeze(self):
        """Mở khóa toàn bộ trọng số MambaVision (`requires_grad = True`) để Fine-tune toàn mạng."""
        self._is_frozen = False
        for param in self.backbone.parameters():
            param.requires_grad = True
        self.backbone.train(self.training)
        return self

    def set_trainable(self, trainable: bool):
        """Công tắc đóng/mở trọng số: True = Mở khóa (Unfreeze), False = Đóng băng (Freeze)."""
        return self.unfreeze() if trainable else self.freeze()

    @property
    def is_frozen(self) -> bool:
        return self._is_frozen

    def train(self, mode: bool = True):
        """
        Đảm bảo khi `trainer.py` gọi `model.train()` ở đầu mỗi Epoch, nếu backbone đang bị
        đóng băng (`_is_frozen=True`), các tầng BatchNorm bên trong `self.backbone` vẫn giữ
        nguyên trạng thái `.eval()`, tránh làm hỏng thống kê Pre-trained.
        """
        super().train(mode)
        if self._is_frozen:
            self.backbone.eval()
        return self

    def forward(self, x):
        x = self.input_adapter(x)
        if self.normalize_imagenet:
            x = (x - self.img_mean) / self.img_std

        # Nếu `input_adapter` có tham số học được thì gradient PHẢI chảy ngược qua backbone (dù backbone
        # đóng băng), nên không được bọc no_grad. Chỉ khi phía trước không có gì để học mới tiết kiệm bộ nhớ.
        adapter_trainable = any(p.requires_grad for p in self.input_adapter.parameters())
        if self._is_frozen and not adapter_trainable:
            with torch.no_grad():
                out = self.backbone(x)
        else:
            out = self.backbone(x)

        return self._extract_feature_list(out)


def _cmd_selftest(_):
    print("Đang chạy kiểm thử tự động custom_backbone.py...")
    enc = MambaVisionEncoder(
        model_name="mambavision_t",
        in_channels=3,
        pretrained=False,
        freeze_backbone=False,
        use_mock=True,
    )
    assert enc.out_channels == [80, 160, 320, 640], f"Sai số kênh: {enc.out_channels}"

    x = torch.randn(2, 3, 128, 128)
    feats = enc(x)
    expected_shapes = [
        (2, 80, 32, 32),
        (2, 160, 16, 16),
        (2, 320, 8, 8),
        (2, 640, 4, 4),
    ]
    for i, (f, exp) in enumerate(zip(feats, expected_shapes)):
        assert tuple(f.shape) == exp, f"Stage {i+1} sai shape: {tuple(f.shape)} != {exp}"

    # Kiểm tra cơ chế Đóng băng (Freeze)
    enc.freeze()
    assert enc.is_frozen is True
    enc.train(True)  # Giả lập trainer gọi model.train()
    assert enc.backbone.training is False, "Khi freeze, backbone bắt buộc phải ở chế độ eval()"
    assert all(not p.requires_grad for p in enc.backbone.parameters())

    # Kiểm tra cơ chế Mở khóa (Unfreeze)
    enc.unfreeze()
    assert enc.is_frozen is False
    assert enc.backbone.training is True
    assert all(p.requires_grad for p in enc.backbone.parameters())

    # --- Các ca hồi quy: lỗi gradient adapter, BN khi freeze, unfreeze khi eval, kích thước lẻ ---
    head = nn.Conv2d(80, 1, 1)
    e5 = MambaVisionEncoder(in_channels=5, pretrained=False, freeze_backbone=True, use_mock=True)
    head(e5(torch.randn(2, 5, 64, 64))[0]).mean().backward()
    g = e5.input_adapter.weight.grad
    assert g is not None and g.abs().sum() > 0, "Freeze + in_channels!=3: input_adapter phải nhận gradient"
    assert all(p.grad is None for p in e5.backbone.parameters()), "Backbone đóng băng không được có gradient"

    ef = MambaVisionEncoder(pretrained=False, freeze_backbone=True, use_mock=True).train()
    before = [m.running_mean.clone() for m in ef.modules() if isinstance(m, nn.BatchNorm2d)]
    ef(torch.randn(2, 3, 64, 64))
    after = [m.running_mean for m in ef.modules() if isinstance(m, nn.BatchNorm2d)]
    assert all(torch.equal(a, b) for a, b in zip(before, after)), "Freeze phải giữ nguyên thống kê BatchNorm"

    ef.eval().unfreeze()
    assert ef.backbone.training is False, "unfreeze() khi module đang eval không được ép backbone về train"

    for size in (96, 100, 512):
        out = MambaVisionEncoder(pretrained=False, use_mock=True)(torch.randn(1, 3, size, size))
        assert len(out) == 4 and out[0].shape[-1] == size // 4

    class _HFLike:  # mô phỏng ModelOutput của Hugging Face
        hidden_states = [torch.zeros(1, c, 2, 2) for c in (80, 160, 320, 640)]
    assert len(MambaVisionEncoder._extract_feature_list(_HFLike())) == 4
    assert [MAMBAVISION_BASE_DIM["mambavision_t"] * 2 ** i for i in range(4)] == enc.out_channels

    print("✅ selftest custom_backbone.py: OK!")

# -*- coding: utf-8 -*-
"""
dune_unet.py - Kiến trúc mạng nơ-ron cho dự án DuneAI.
Hỗ trợ đầu vào 2.5D (3 kênh: lát z-1, z, z+1) và đầu ra 1 kênh (mặt nạ khối u).
Được thiết kế theo dạng Module để dễ dàng cắm rút và nâng cấp lên Mamba-UNet.
"""
import torch
import torch.nn as nn
import torch.nn.functional as F

class DoubleConv(nn.Module):
    """Khối tích chập kép cơ bản của U-Net: Conv2d -> BatchNorm -> ReLU -> Conv2d -> BatchNorm -> ReLU"""
    def __init__(self, in_channels, out_channels, mid_channels=None):
        super().__init__()
        if not mid_channels:
            mid_channels = out_channels
        self.double_conv = nn.Sequential(
            nn.Conv2d(in_channels, mid_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(mid_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(mid_channels, out_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True)
        )

    def forward(self, x):
        return self.double_conv(x)


class Down(nn.Module):
    """Khối thu nhỏ (Encoder): MaxPool2d để giảm kích thước ảnh một nửa, sau đó dùng DoubleConv"""
    def __init__(self, in_channels, out_channels):
        super().__init__()
        self.maxpool_conv = nn.Sequential(
            nn.MaxPool2d(2),
            DoubleConv(in_channels, out_channels)
        )

    def forward(self, x):
        return self.maxpool_conv(x)


class Up(nn.Module):
    """Khối phóng to (Decoder) và kết nối chéo (Skip Connection)"""
    def __init__(self, in_channels, out_channels):
        super().__init__()
        # Dùng ConvTranspose2d để phóng to ảnh lên gấp đôi
        self.up = nn.ConvTranspose2d(in_channels, in_channels // 2, kernel_size=2, stride=2)
        self.conv = DoubleConv(in_channels, out_channels)

    def forward(self, x1, x2):
        # Phóng to x1 (từ lớp dưới lên)
        x1 = self.up(x1)
        
        # Xử lý trường hợp kích thước ảnh bị lẻ (padding để khớp với x2)
        diffY = x2.size()[2] - x1.size()[2]
        diffX = x2.size()[3] - x1.size()[3]
        x1 = F.pad(x1, [diffX // 2, diffX - diffX // 2,
                        diffY // 2, diffY - diffY // 2])
        
        # Kết nối chéo: Nối ma trận đặc trưng x2 (từ Encoder) với x1 (vừa phóng to)
        x = torch.cat([x2, x1], dim=1)
        return self.conv(x)


class OutConv(nn.Module):
    """Lớp xuất cuối cùng: Chuyển số lượng kênh đặc trưng về số lượng lớp nhãn (1 kênh)"""
    def __init__(self, in_channels, out_channels):
        super(OutConv, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size=1)

    def forward(self, x):
        return self.conv(x)


class DuneUNet(nn.Module):
    def __init__(self, in_channels=3, classes=1, features=64):
        """
        Khởi tạo mạng DuneUNet cho chuẩn 2.5D.
        - in_channels: 3 (Lát cắt Z-1, Z, Z+1).
        - classes: 1 (Dự đoán pixel có phải là khối u hay không).
        - features: Số lượng bộ lọc (filter) cơ sở. Càng lớn mạng càng thông minh nhưng ngốn VRAM.
        """
        super(DuneUNet, self).__init__()
        self.in_channels = in_channels
        self.classes = classes

        # 1. ENCODER (Khối chiết xuất đặc trưng - Dần dần thu nhỏ ảnh, tăng số kênh)
        self.inc = DoubleConv(in_channels, features)
        self.down1 = Down(features, features * 2)
        self.down2 = Down(features * 2, features * 4)
        self.down3 = Down(features * 4, features * 8)
        
        # 2. BOTTLENECK (Đáy chữ U - Điểm sâu nhất của mạng)
        # NƠI CẤY GHÉP MAMBA: Sau này bạn có thể thay thế DoubleConv ở đây bằng các khối MambaVision 
        # để tăng khả năng nắm bắt bối cảnh toàn cục (Global Context).
        self.down4 = Down(features * 8, features * 16)
        
        # 3. DECODER (Khối giải mã - Dần dần phóng to ảnh, giảm số kênh)
        self.up1 = Up(features * 16, features * 8)
        self.up2 = Up(features * 8, features * 4)
        self.up3 = Up(features * 4, features * 2)
        self.up4 = Up(features * 2, features)
        
        # Lớp đầu ra (Logits)
        self.outc = OutConv(features, classes)
        
        # Khởi tạo bias âm để mô hình bắt đầu với xác suất nền cao (u rất nhỏ so với ảnh)
        nn.init.constant_(self.outc.conv.bias, -4.0)

    def forward(self, x):
        # Đi xuống (Encoder)
        x1 = self.inc(x)
        x2 = self.down1(x1)
        x3 = self.down2(x2)
        x4 = self.down3(x3)
        
        # Đáy (Bottleneck)
        x5 = self.down4(x4)
        
        # Đi lên (Decoder + Skip Connections)
        x = self.up1(x5, x4)
        x = self.up2(x, x3)
        x = self.up3(x, x2)
        x = self.up4(x, x1)
        
        logits = self.outc(x)
        return logits

def build_dataset_index(

    zero_sl_ratio=ZERO_SL_RATIO,
    val_split=VAL_SPLIT,
    seed=SEED,
    require_lung=True,
):
    """
    Quét thư mục cache_1mm, lọc lát cắt theo tiêu chuẩn y khoa của DuneAI và xuất slices.csv.
    Lưu ý: mask_1mm.npy đã ở tỷ lệ 1mm/pixel nên 1 pixel = 1.0 mm^2.
    """
    cache_dir = Path(cache_dir)
    if out_csv_path is None:
        out_csv_path = cache_dir / "slices.csv"
    out_csv_path = Path(out_csv_path)

    if not cache_dir.exists():
        raise FileNotFoundError(f"Không tìm thấy thư mục cache: {cache_dir}")

    patient_ids = sorted([
        p.name for p in cache_dir.iterdir()
        if p.is_dir() and (p / "meta.json").exists() and (p / "image_1mm.npy").exists()
    ])
    if not patient_ids:
        raise RuntimeError(f"Không tìm thấy bệnh nhân hợp lệ nào trong {cache_dir}")

    # Chia Train/Val theo cấp độ Bệnh nhân (Patient-level split) để chống Data Leakage
    rng = random.Random(seed)
    shuffled_pids = list(patient_ids)
    rng.shuffle(shuffled_pids)

    if len(shuffled_pids) == 1:
        val_patients = set()
        train_patients = set(shuffled_pids)
    else:
        val_size = max(1, int(round(len(shuffled_pids) * val_split)))
        val_patients = set(shuffled_pids[:val_size])
        train_patients = set(shuffled_pids[val_size:])

    rows = []
    stats = {
        "patients_total": len(patient_ids),
        "patients_train": len(train_patients),
        "patients_val": len(val_patients),
        "dropped_no_lung": 0,
        "dropped_small_tumor": 0,
        "dropped_neg_sampled": 0,
    }

    for pid in patient_ids:
        p_dir = cache_dir / pid

def unpad_or_uncrop_xy(x_padded, target_hw, pad_value=0):
    """
    Hàm nghịch đảo chính xác tuyệt đối của `pad_or_crop_xy` trong `medical_prep.py`.
    Chuyển mảng/tensor từ kích thước chuẩn mạng (..., IMAGE_SIZE, IMAGE_SIZE)
    về lại kích thước vật lý 1mm của bệnh nhân (..., H_1mm, W_1mm).
    """
    size_h, size_w = x_padded.shape[-2], x_padded.shape[-1]
    assert size_h == size_w, f"Đầu vào phải có mặt phẳng vuông, nhận được: {x_padded.shape}"
    size = size_h
    h, w = int(target_hw[0]), int(target_hw[1])

    sy, dy, ly = _axis_window(h, size)
    sx, dx, lx = _axis_window(w, size)
    shape = (*x_padded.shape[:-2], h, w)

    if isinstance(x_padded, torch.Tensor):
        out = torch.full(shape, pad_value, dtype=x_padded.dtype, device=x_padded.device)
    else:
        out = np.full(shape, pad_value, dtype=x_padded.dtype)

    out[..., sy : sy + ly, sx : sx + lx] = x_padded[..., dy : dy + ly, dx : dx + lx]
    return out


def resize_from_1mm_3d(volume_1mm, orig_shape_zyx, is_mask=True):
    """
    Đưa khối 3D từ lưới 1mm (Z, H_1mm, W_1mm) về lại độ phân giải gốc (Z, H_orig, W_orig)
    của file NRRD/DICOM ban đầu để xuất file cho bác sĩ hoặc đối chiếu trên 3D Slicer / Viewer.
    """
    z_orig, h_orig, w_orig = (int(v) for v in orig_shape_zyx)
    z, h, w = volume_1mm.shape
    assert z == z_orig, f"Số lát cắt Z không khớp: {z} != {z_orig}"

    if h == h_orig and w == w_orig:
        return volume_1mm.copy()

    flag = cv2.INTER_NEAREST if is_mask else cv2.INTER_LINEAR
    src = (volume_1mm > 0).astype(np.uint8) if is_mask else np.ascontiguousarray(volume_1mm)
    out = np.empty((z_orig, h_orig, w_orig), dtype=src.dtype)
    for i in range(z_orig):
        out[i] = cv2.resize(src[i], (w_orig, h_orig), interpolation=flag)
    return out


# ======================================================================
# 2. HẬU XỬ LÝ HÌNH THÁI HỌC & LỌC LIÊN THÔNG 3D (3D POST-PROCESSING)
# ======================================================================
def postprocess_3d_mask(
    prob_or_mask_zyx,
    lung_mask_zyx=None,
    threshold=0.5,
    min_slice_area_mm2=DEFAULT_TUMOR_LOW_LIM,
    min_volume_mm3=0.0,
    spacing_z=1.0,
    fill_holes=True,
    min_slices_span=1,
    top_k_components=0,
):
    """
    Hậu xử lý khối dự đoán 3D trên lưới 1mm (1 pixel trên mặt phẳng XY = 1.0 mm²):
      1. Ngưỡng hóa xác suất -> nhị phân (uint8).
      2. Cắt bỏ mọi dự đoán nằm ngoài mặt nạ phổi (nếu có lung_mask_zyx).
      3. Lấp các lỗ khuyết nhỏ bên trong tâm khối u trên từng lát cắt 2D (binary_fill_holes).
      4. Phân tích thành phần liên thông 3D (26-connectivity):
         - Loại bỏ cụm nhiễu có diện tích mặt cắt ngang lớn nhất < min_slice_area_mm2 (20.0 mm² theo chuẩn tác giả).
         - Loại bỏ cụm có thể tích vật lý < min_volume_mm3 hoặc chỉ xuất hiện ít hơn min_slices_span lát cắt.
         - Tùy chọn chỉ giữ lại top_k_components khối lớn nhất.
    """
    arr = np.asarray(prob_or_mask_zyx)
    if np.issubdtype(arr.dtype, np.floating):
        mask = (arr >= float(threshold)).astype(np.uint8)
    else:
        mask = (arr > 0).astype(np.uint8)

    if lung_mask_zyx is not None:
        keep_lung = lung_keep_mask(lung_mask_zyx, tumor_mask=None)
        mask &= keep_lung

    if mask.sum() == 0:
        return mask

    # Lấp lỗ rỗng 2D trên từng lát cắt có dự đoán dương tính
    if fill_holes:
        for z in np.where(mask.sum(axis=(1, 2)) > 0)[0]:
            mask[z] = ndi.binary_fill_holes(mask[z]).astype(np.uint8)

    # Đánh nhãn các khối liên thông 3D (26-connectivity: khối lập phương 3x3x3)
    structure_3d = np.ones((3, 3, 3), dtype=np.uint8)
    labeled, num_features = ndi.label(mask, structure=structure_3d)
    if num_features == 0:
        return np.zeros_like(mask, dtype=np.uint8)

    objects = ndi.find_objects(labeled)
    valid_components = []

    for comp_id, slc in enumerate(objects, start=1):
        if slc is None:
            continue
        sub_mask = (labeled[slc] == comp_id)
        voxel_count = int(sub_mask.sum())
        if voxel_count == 0:
            continue

        # Số lát cắt Z mà cụm này trải qua
        z_span = int(sub_mask.shape[0])
        if z_span < min_slices_span:
            continue

        # Diện tích mặt cắt ngang cực đại (ở lưới 1mm, 1 pixel = 1.0 mm²)
        slice_areas = sub_mask.reshape(z_span, -1).sum(axis=1)
        max_area_mm2 = float(slice_areas.max())
        if max_area_mm2 < float(min_slice_area_mm2):
            continue

        # Thể tích vật lý 3D (mm³) = số voxel * (1mm * 1mm * spacing_z)
        vol_mm3 = float(voxel_count) * float(spacing_z)
        if vol_mm3 < float(min_volume_mm3):
            continue

        valid_components.append((comp_id, voxel_count))

    if not valid_components:
        return np.zeros_like(mask, dtype=np.uint8)

    if top_k_components and top_k_components > 0 and len(valid_components) > top_k_components:
        valid_components.sort(key=lambda x: x[1], reverse=True)
        valid_components = valid_components[:top_k_components]

    keep_ids = np.array([cid for cid, _ in valid_components], dtype=labeled.dtype)
    cleaned_mask = np.isin(labeled, keep_ids).astype(np.uint8)
    return cleaned_mask


# ======================================================================
# 3. CHỈ SỐ ĐÁNH GIÁ 3D CẤP ĐỘ BỆNH NHÂN (3D PATIENT-LEVEL METRICS)
# ======================================================================
def _compute_hd95_mm(pred_bin, gt_bin, spacing_zyx=(1.0, 1.0, 1.0)):
    """
    Tính 95th percentile Hausdorff Distance (HD95) theo đơn vị mm vật lý giữa 2 mặt nạ 3D.
    Nếu một trong hai rỗng: trả về 0.0 khi cả hai cùng rỗng, hoặc NaN khi chỉ một bên rỗng.
    """
    pred_any = bool(pred_bin.any())
    gt_any = bool(gt_bin.any())
    if not pred_any and not gt_any:
        return 0.0
    if not pred_any or not gt_any:
        return float("nan")

    # Trích xuất biên bề mặt 3D bằng phép co (erosion)
    pred_border = pred_bin ^ ndi.binary_erosion(pred_bin)
    gt_border = gt_bin ^ ndi.binary_erosion(gt_bin)
    if not pred_border.any():
        pred_border = pred_bin
    if not gt_border.any():
        gt_border = gt_bin

    dt_gt = ndi.distance_transform_edt(~gt_border, sampling=spacing_zyx)
    dt_pred = ndi.distance_transform_edt(~pred_border, sampling=spacing_zyx)

    dists_pred_to_gt = dt_gt[pred_border]
    dists_gt_to_pred = dt_pred[gt_border]
    all_dists = np.concatenate([dists_pred_to_gt, dists_gt_to_pred])
    return float(np.percentile(all_dists, 95))


def compute_3d_metrics(pred_mask, gt_mask, spacing_z=1.0, compute_hd95=True, eps=1e-6):
    """
    Tính bộ chỉ số phân vùng 3D trên toàn bộ thể tích phổi của bệnh nhân (lưới 1mm):
      - dice_3d, iou_3d, precision_3d, recall_3d
      - pred_volume_mm3, gt_volume_mm3
      - hd95_mm (Hausdorff Distance 95% tính bằng mm)
    """
    pred_b = np.asarray(pred_mask) > 0
    gt_b = np.asarray(gt_mask) > 0

    tp = float(np.logical_and(pred_b, gt_b).sum())
    fp = float(np.logical_and(pred_b, ~gt_b).sum())
    fn = float(np.logical_and(~pred_b, gt_b).sum())

    pred_sum = tp + fp
    gt_sum = tp + fn
    union = tp + fp + fn

    # Trường hợp ca bệnh hoàn toàn không có u và AI cũng dự đoán không có u -> đúng tuyệt đối (1.0)
    if pred_sum == 0 and gt_sum == 0:
        dice = 1.0
        iou = 1.0
        precision = 1.0
        recall = 1.0
    else:
        dice = (2.0 * tp + eps) / (pred_sum + gt_sum + eps)
        iou = (tp + eps) / (union + eps)
        precision = (tp + eps) / (pred_sum + eps) if pred_sum > 0 else 0.0
        recall = (tp + eps) / (gt_sum + eps) if gt_sum > 0 else 0.0

    hd95 = _compute_hd95_mm(pred_b, gt_b, spacing_zyx=(float(spacing_z), 1.0, 1.0)) if compute_hd95 else float("nan")

    return {
        "dice_3d": float(dice),
        "iou_3d": float(iou),
        "precision_3d": float(precision),
        "recall_3d": float(recall),
        "pred_voxels": int(pred_sum),
        "gt_voxels": int(gt_sum),
        "pred_volume_mm3": round(float(pred_sum * spacing_z), 2),
        "gt_volume_mm3": round(float(gt_sum * spacing_z), 2),
        "hd95_mm": round(float(hd95), 3) if np.isfinite(hd95) else float("nan"),
    }

# 4. SUY LUẬN 3D TOÀN BỆNH NHÂN (2.5D SLIDING-WINDOW RECONSTRUCTION)
# ======================================================================
def load_trained_model(checkpoint_path, device=None, default_features=64, in_channels=3):
    """
    Nạp mô hình DuneUNet từ file checkpoint (.pth).
    Tự động suy luận số `features` (32, 64, ...) từ kích thước trọng số `inc.double_conv.0.weight`
    để người dùng không bao giờ bị lỗi lệch số kênh khi load model.
    """
    device = device or torch.device("cuda" if torch.cuda.is_available() else "cpu")
    ckpt = torch.load(checkpoint_path, map_location=device)
    state = ckpt["model_state_dict"] if isinstance(ckpt, dict) and "model_state_dict" in ckpt else ckpt

    # Bóc tiền tố 'module.' nếu checkpoint từng lưu trực tiếp từ DataParallel
    cleaned_state = {}
    for k, v in state.items():
        cleaned_state[k[7:] if k.startswith("module.") else k] = v

    first_conv_key = "inc.double_conv.0.weight"
    if first_conv_key in cleaned_state:
        features = int(cleaned_state[first_conv_key].shape[0])
        in_channels = int(cleaned_state[first_conv_key].shape[1])
    else:
        features = default_features

    model = DuneUNet(in_channels=in_channels, classes=1, features=features)
    model.load_state_dict(cleaned_state)
    model.to(device)
    model.eval()
    return model


@torch.no_grad()
def predict_patient_volume(
    model,
    patient_cache_dir,
    device=None,
    batch_size=8,
    in_channels=3,
    image_size=IMAGE_SIZE,
    apply_lung=True,
    threshold=0.5,
    min_slice_area_mm2=DEFAULT_TUMOR_LOW_LIM,
    min_volume_mm3=0.0,
    tta=False,
    use_amp=True,
):
    """
    Suy luận 3D hoàn chỉnh cho 1 bệnh nhân từ thư mục cache_1mm/<patient_id>:
      1. Đọc image_1mm.npy, mask_1mm.npy, lung_1mm.npy, meta.json.
      2. Áp mặt nạ phổi (KHÔNG dùng tumor_mask để chống rò rỉ nhãn).
      3. Chỉ chạy mạng U-Net trên các lát cắt nằm trong vùng có phổi (tiết kiệm 35-50% thời gian GPU).
      4. Khôi phục kích thước từ (Z, 512, 512) về (Z, H_1mm, W_1mm) bằng `unpad_or_uncrop_xy`.
      5. Chạy hậu xử lý 3D (`postprocess_3d_mask`) để lọc nhiễu và tính Dice 3D trước/sau hậu xử lý.
    """
    device = device or next(model.parameters()).device
    is_cuda = device.type == "cuda"
    model.eval()

    patient_cache_dir = Path(patient_cache_dir)
    img_1mm, msk_1mm, meta = load_cached(patient_cache_dir, mmap=True)
    num_slices, h_1mm, w_1mm = img_1mm.shape
    spacing_z = float(meta.get("orig_spacing_xyz", [1.0, 1.0, 1.0])[2])

    lung_p = lung_cache_path(patient_cache_dir)
    has_lung = apply_lung and lung_p.exists() and (meta.get("lung_ok", True) is not False)
    lung_1mm = np.asarray(np.load(lung_p, mmap_mode="r")) if has_lung else None

    vol_hu = np.asarray(img_1mm)
    if lung_1mm is not None:
        keep_lung = lung_keep_mask(lung_1mm, tumor_mask=None)
        vol_hu = apply_lung_mask(vol_hu, keep_lung, fill=HU_FLOOR)
        active_z = np.where(keep_lung.reshape(num_slices, -1).sum(axis=1) > 0)[0].tolist()
    else:
        keep_lung = None
        active_z = list(range(num_slices))

    # Pad/crop toàn bộ khối về (Z, image_size, image_size) để trích xuất nhanh các lát 2.5D
    vol_padded = pad_or_crop_xy(vol_hu, size=image_size, pad_value=HU_FLOOR)
    prob_padded = np.zeros((num_slices, image_size, image_size), dtype=np.float32)

    half_k = in_channels // 2
    amp_ctx = (
        torch.amp.autocast("cuda", enabled=True)
        if (use_amp and is_cuda and hasattr(torch, "amp") and hasattr(torch.amp, "autocast"))
        else nullcontext()
    )

    # Suy luận theo batch trên các lát cắt chứa nhu mô phổi
    for start in range(0, len(active_z), batch_size):
        batch_z = active_z[start : start + batch_size]
        stacks = []
        for z in batch_z:
            z_idx = [min(max(0, z + off), num_slices - 1) for off in range(-half_k, half_k + 1)]
            stacks.append(vol_padded[z_idx])

        batch_hu = torch.from_numpy(np.stack(stacks, axis=0)).to(device, non_blocking=is_cuda)
        batch_x = window_hu_torch(batch_hu)

        with amp_ctx:
            logits = model(batch_x)
            probs = torch.sigmoid(logits)
            if tta:
                # Test-Time Augmentation: trung bình cộng với dự đoán lật ngang và lật dọc
                logits_h = torch.flip(model(torch.flip(batch_x, dims=[3])), dims=[3])
                logits_v = torch.flip(model(torch.flip(batch_x, dims=[2])), dims=[2])
                probs = (probs + torch.sigmoid(logits_h) + torch.sigmoid(logits_v)) / 3.0

        probs_np = probs.squeeze(1).to(torch.float32).cpu().numpy()
        for i, z in enumerate(batch_z):
            prob_padded[z] = probs_np[i]

    # Khôi phục từ (Z, image_size, image_size) về đúng lưới 1mm (Z, H_1mm, W_1mm)
    prob_1mm = unpad_or_uncrop_xy(prob_padded, (h_1mm, w_1mm), pad_value=0.0)
    raw_mask_1mm = (prob_1mm >= float(threshold)).astype(np.uint8)
    if keep_lung is not None:
        raw_mask_1mm &= keep_lung

    # Hậu xử lý hình thái học 3D
    clean_mask_1mm = postprocess_3d_mask(
        prob_1mm,
        lung_mask_zyx=keep_lung,
        threshold=threshold,
        min_slice_area_mm2=min_slice_area_mm2,
        min_volume_mm3=min_volume_mm3,
        spacing_z=spacing_z,
        fill_holes=True,
    )

    gt_1mm = (np.asarray(msk_1mm) > 0).astype(np.uint8)
    metrics_raw = compute_3d_metrics(raw_mask_1mm, gt_1mm, spacing_z=spacing_z, compute_hd95=False)
    metrics_post = compute_3d_metrics(clean_mask_1mm, gt_1mm, spacing_z=spacing_z, compute_hd95=True)

    return {
        "prob_1mm": prob_1mm,
        "raw_mask_1mm": raw_mask_1mm,
        "clean_mask_1mm": clean_mask_1mm,
        "gt_mask_1mm": gt_1mm,
        "meta": meta,
        "metrics_raw": metrics_raw,
        "metrics_post": metrics_post,
    }


def save_prediction_nrrd(clean_mask_1mm, meta, out_nrrd_path, reference_nrrd_path=None):
    """
    Đưa mặt nạ dự đoán từ lưới 1mm (Z, H_1mm, W_1mm) về lưới gốc (Z, H_orig, W_orig)
    và lưu thành file `.nrrd` giữ nguyên hệ tọa độ y khoa (Spacing, Origin, Direction).
    """
    if sitk is None or not hasattr(sitk, "GetImageFromArray"):
        raise ImportError("Cần cài đặt SimpleITK để xuất file NRRD: pip install SimpleITK")

    orig_shape_zyx = meta.get("orig_shape_zyx", list(clean_mask_1mm.shape))
    mask_orig = resize_from_1mm_3d(clean_mask_1mm, orig_shape_zyx, is_mask=True)

    sitk_mask = sitk.GetImageFromArray(mask_orig.astype(np.uint8))
    if reference_nrrd_path and Path(reference_nrrd_path).exists():
        ref_img = sitk.ReadImage(str(reference_nrrd_path))
        sitk_mask.CopyInformation(ref_img)
    elif "orig_spacing_xyz" in meta:
        sitk_mask.SetSpacing(tuple(float(s) for s in meta["orig_spacing_xyz"]))

    out_nrrd_path = Path(out_nrrd_path)
    out_nrrd_path.parent.mkdir(parents=True, exist_ok=True)
    sitk.WriteImage(sitk_mask, str(out_nrrd_path), useCompression=True)
    return mask_orig

# 5. ĐÁNH GIÁ TOÀN TẬP DỮ LIỆU (PATIENT-LEVEL 3D EVALUATION)
# ======================================================================
def evaluate_dataset_3d(
    cache_dir,
    checkpoint_path,
    split="val",
    batch_size=8,
    threshold=0.5,
    min_slice_area_mm2=DEFAULT_TUMOR_LOW_LIM,
    tta=False,
    out_csv=None,
):
    """
    Chạy suy luận 3D và hậu xử lý cho toàn bộ bệnh nhân thuộc `split` ('val', 'train', hoặc 'all').
    So sánh trực tiếp Dice 3D trước và sau hậu xử lý.
    """
    cache_dir = Path(cache_dir)
    csv_path = cache_dir / "slices.csv"
    if csv_path.exists() and split != "all":
        df_slices = pd.read_csv(csv_path)
        pids = sorted(df_slices[df_slices["split"] == split]["patient_id"].unique().tolist())
    else:
        pids = sorted([
            p.name for p in cache_dir.iterdir()
            if p.is_dir() and (p / "meta.json").exists() and (p / "image_1mm.npy").exists()
        ])

    if not pids:
        raise RuntimeError(f"Không tìm thấy bệnh nhân nào cho split='{split}' trong {cache_dir}")

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = load_trained_model(checkpoint_path, device=device)

    print("=" * 90)
    print(
        f"🩻 ĐÁNH GIÁ 3D CẤP ĐỘ BỆNH NHÂN | Split: {split} ({len(pids)} ca) | "
        f"Thiết bị: {device} | Ngưỡng lọc u: >= {min_slice_area_mm2} mm²"
    )
    print("=" * 90)

    rows = []
    for idx, pid in enumerate(pids, start=1):
        t0 = time.time()
        res = predict_patient_volume(
            model=model,
            patient_cache_dir=cache_dir / pid,
            device=device,
            batch_size=batch_size,
            threshold=threshold,
            min_slice_area_mm2=min_slice_area_mm2,
            tta=tta,
        )
        dt = time.time() - t0
        m_raw = res["metrics_raw"]
        m_post = res["metrics_post"]

        row = {
            "patient_id": pid,
            "dice_3d_raw": round(m_raw["dice_3d"], 4),
            "dice_3d_post": round(m_post["dice_3d"], 4),
            "iou_3d_post": round(m_post["iou_3d"], 4),
            "precision_3d": round(m_post["precision_3d"], 4),
            "recall_3d": round(m_post["recall_3d"], 4),
            "hd95_mm": m_post["hd95_mm"],
            "gt_vol_mm3": m_post["gt_volume_mm3"],
            "pred_vol_mm3": m_post["pred_volume_mm3"],
            "time_sec": round(dt, 2),
        }
        rows.append(row)
        diff_dice = m_post["dice_3d"] - m_raw["dice_3d"]
        print(
            f"[{idx:02d}/{len(pids):02d}] {pid} ({dt:.1f}s) | "
            f"3D Dice: {m_raw['dice_3d']:.4f} -> {m_post['dice_3d']:.4f} ({diff_dice:+.4f}) | "
            f"IoU: {m_post['iou_3d']:.4f} | HD95: {m_post['hd95_mm']} mm | "
            f"Vol(GT/Pred): {m_post['gt_volume_mm3']}/{m_post['pred_volume_mm3']} mm³"
        )

    df_res = pd.DataFrame(rows)
    mean_raw = float(df_res["dice_3d_raw"].mean())
    mean_post = float(df_res["dice_3d_post"].mean())
    mean_iou = float(df_res["iou_3d_post"].mean())

    print("=" * 90)
    print(
        f"📊 TỔNG KẾT ({len(pids)} bệnh nhân) | "
        f"Mean 3D Dice (Raw): {mean_raw:.4f} -> Mean 3D Dice (Post): {mean_post:.4f} "
        f"({mean_post - mean_raw:+.4f}) | Mean 3D IoU: {mean_iou:.4f}"
    )
    print("=" * 90)

    if out_csv:
        out_csv = Path(out_csv)
        out_csv.parent.mkdir(parents=True, exist_ok=True)
        df_res.to_csv(out_csv, index=False)
        print(f"✅ Đã lưu bảng báo cáo đánh giá 3D tại: {out_csv}")

    return df_res

def _cmd_selftest(_):
    print("Đang chạy kiểm thử tự động postprocess.py (Inverse Geometry + 3D Filtering + 3D Pipeline)...")

    # 1. Kiểm thử tính nghịch đảo hoàn hảo của unpad_or_uncrop_xy và resize_from_1mm_3d
    for hw in [(360, 380), (512, 512), (580, 600)]:
        arr = np.zeros((4, hw[0], hw[1]), dtype=np.uint8)
        ch, cw = hw[0] // 2, hw[1] // 2
        arr[1:3, ch - 20 : ch + 20, cw - 20 : cw + 20] = 1
        padded = pad_or_crop_xy(arr, size=512, pad_value=0)
        unpadded = unpad_or_uncrop_xy(padded, hw, pad_value=0)
        assert np.array_equal(arr, unpadded), f"Lỗi nghịch đảo pad/crop với kích thước {hw}"

    orig_back = resize_from_1mm_3d(np.ones((4, 360, 360), dtype=np.uint8), (4, 512, 512), is_mask=True)
    assert orig_back.shape == (4, 512, 512) and set(np.unique(orig_back)) <= {0, 1}

    # 2. Kiểm thử bộ lọc hình thái học 3D (loại bỏ nhiễu < 20mm², nhiễu ngoài phổi, và lấp lỗ rỗng)
    gt = np.zeros((10, 128, 128), dtype=np.uint8)
    lung = np.zeros((10, 128, 128), dtype=np.uint8)
    lung[1:9, 10:118, 10:118] = 1

    # Khối u thật: 10x10 = 100 mm² mỗi lát trên các lát 4, 5, 6 (tổng 300 voxels)
    gt[4:7, 50:60, 50:60] = 1

    noisy_prob = gt.astype(np.float32) * 0.95
    # Tạo lỗ thủng ở giữa khối u để kiểm tra binary_fill_holes
    noisy_prob[5, 54:56, 54:56] = 0.1
    # Thêm đốm dương tính giả nhỏ trong phổi: 3x3 = 9 mm² (< 20 mm²) ở lát 2 -> phải bị lọc bỏ!
    noisy_prob[2, 30:33, 30:33] = 0.90
    # Thêm đám nhiễu to nằm NGOÀI phổi (lát 0 và góc biên) -> phải bị mặt nạ phổi cắt bỏ!
    noisy_prob[0, 40:60, 40:60] = 0.95
    noisy_prob[5, 0:8, 0:8] = 0.95

    raw_m = compute_3d_metrics(noisy_prob >= 0.5, gt, spacing_z=1.5)
    cleaned = postprocess_3d_mask(
        noisy_prob, lung_mask_zyx=lung, threshold=0.5, min_slice_area_mm2=20.0, spacing_z=1.5, fill_holes=True
    )
    post_m = compute_3d_metrics(cleaned, gt, spacing_z=1.5)

    assert np.array_equal(cleaned, gt), "Hậu xử lý 3D phải khôi phục chính xác 100% khối u gốc!"
    assert post_m["dice_3d"] > 0.999 and post_m["hd95_mm"] == 0.0
    assert raw_m["dice_3d"] < 0.70, f"Dice trước hậu xử lý phải thấp do nhiễu, nhận được {raw_m['dice_3d']}"

    # 3. Kiểm thử tích hợp toàn trình: Load Checkpoint -> Suy luận 2.5D trên Cache -> Hậu xử lý 3D
    tmp = Path(tempfile.mkdtemp())
    try:
        p_dir = tmp / "LIDC-IDRI-0001"
        p_dir.mkdir(parents=True)
        img = np.full((8, 100, 100), -800, dtype=np.int16)
        msk = np.zeros((8, 100, 100), dtype=np.uint8)
        lung_c = np.zeros((8, 100, 100), dtype=np.uint8)
        lung_c[1:7, 10:90, 10:90] = 1
        msk[3:5, 40:50, 40:50] = 1
        img[3:5, 40:50, 40:50] = 50

        np.save(p_dir / "image_1mm.npy", img)
        np.save(p_dir / "mask_1mm.npy", msk)
        np.save(p_dir / "lung_1mm.npy", lung_c)
        with open(p_dir / "meta.json", "w", encoding="utf-8") as f:
            json.dump({
                "orig_shape_zyx": [8, 128, 128],
                "orig_spacing_xyz": [0.78, 0.78, 1.5],
                "shape_1mm_zyx": [8, 100, 100],
                "lung_ok": True,
            }, f)

        model = DuneUNet(in_channels=3, classes=1, features=8)
        ckpt_path = tmp / "dummy_model.pth"
        torch.save({"model_state_dict": model.state_dict()}, ckpt_path)

        loaded_model = load_trained_model(ckpt_path, device=torch.device("cpu"))
        res = predict_patient_volume(
            loaded_model, p_dir, batch_size=2, image_size=128, tta=True, use_amp=False
        )
        assert res["prob_1mm"].shape == (8, 100, 100)
        assert res["clean_mask_1mm"].shape == (8, 100, 100)
        assert "dice_3d" in res["metrics_post"]

        print(
            f"✅ selftest postprocess.py: OK! "
            f"(3D Noise Filter: Raw Dice {raw_m['dice_3d']:.4f} -> Clean Dice {post_m['dice_3d']:.4f}, HD95={post_m['hd95_mm']} mm)"
        )
    finally:
        shutil.rmtree(tmp, ignore_errors=True)


def _cmd_predict(a):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = load_trained_model(a.checkpoint, device=device)
    res = predict_patient_volume(
        model=model,
        patient_cache_dir=a.patient_cache,
        device=device,
        batch_size=a.batch_size,
        threshold=a.threshold,
        min_slice_area_mm2=a.min_area,
        tta=a.tta,
    )
    m_raw, m_post = res["metrics_raw"], res["metrics_post"]
    print("=" * 75)
    print(f"🩻 KẾT QUẢ SUY LUẬN 3D: {Path(a.patient_cache).name}")
    print("=" * 75)
    print(f"  - 3D Dice (Trước hậu xử lý): {m_raw['dice_3d']:.4f} ({m_raw['pred_volume_mm3']} mm³)")
    print(f"  - 3D Dice (Sau hậu xử lý)  : {m_post['dice_3d']:.4f} ({m_post['pred_volume_mm3']} mm³)")
    print(f"  - Thể tích khối u thực (GT): {m_post['gt_volume_mm3']} mm³ | HD95: {m_post['hd95_mm']} mm")
    if a.out_nrrd:
        save_prediction_nrrd(res["clean_mask_1mm"], res["meta"], a.out_nrrd, a.ref_nrrd)
        print(f"💾 Đã xuất file mặt nạ 3D NRRD tại: {a.out_nrrd}")

def format_duration(seconds):
    """Định dạng thời gian (giây) sang chuỗi dễ đọc: 45.2s, 12m 30s, hoặc 02h 15m 10s."""
    if seconds is None or not np.isfinite(seconds) or seconds < 0:
        return "--:--"
    if seconds < 60:
        return f"{seconds:.1f}s"
    sec = int(round(seconds))
    m, s = divmod(sec, 60)
    if m < 60:
        return f"{m:02d}m {s:02d}s"
    h, m = divmod(m, 60)
    return f"{h:02d}h {m:02d}m {s:02d}s"


def _make_grad_scaler(enabled):
    if hasattr(torch, "amp") and hasattr(torch.amp, "GradScaler"):
        try:
            return torch.amp.GradScaler("cuda", enabled=enabled)
        except TypeError:
            pass
    return torch.cuda.amp.GradScaler(enabled=enabled)


def _autocast_context(enabled):
    if not enabled:
        return nullcontext()
    if hasattr(torch, "amp") and hasattr(torch.amp, "autocast"):
        return torch.amp.autocast("cuda", enabled=True)
    return torch.cuda.amp.autocast(enabled=True)

class Trainer:
    def __init__(
        self,
        model,
        criterion,
        optimizer,
        scheduler=None,
        device=None,
        use_amp=True,
        save_dir="checkpoints",
        grad_clip=1.0,
        show_progress=True,
    ):
        self.device = device or torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.is_cuda = self.device.type == "cuda"

        # Cơ chế thích ứng tự động 1 GPU / Nhiều GPU
        if self.is_cuda and torch.cuda.device_count() > 1 and not isinstance(model, nn.DataParallel):
            print(f"🔥 Phát hiện {torch.cuda.device_count()} GPUs! Kích hoạt chế độ song song nn.DataParallel.")
            model = nn.DataParallel(model)

        self.model = model.to(self.device)
        self.criterion = criterion.to(self.device)
        self.optimizer = optimizer
        self.scheduler = scheduler
        self.use_amp = bool(use_amp and self.is_cuda)
        self.scaler = _make_grad_scaler(self.use_amp)

        self.save_dir = Path(save_dir)
        self.save_dir.mkdir(parents=True, exist_ok=True)
        self.grad_clip = grad_clip
        self.show_progress = bool(show_progress)

        # Trạng thái phục vụ huấn luyện và Resume
        self.start_epoch = 1
        self.best_val_dice = -1.0
        self.no_improve = 0
        self.history = []

    def _unwrap_model(self):
        """Bóc lớp bọc DataParallel để lưu state_dict chuẩn, nạp lại được trên mọi máy."""
        return self.model.module if isinstance(self.model, nn.DataParallel) else self.model

    def _prepare_batch(self, batch):
        images = batch["image"].to(self.device, non_blocking=self.is_cuda)
        masks = batch["mask"].to(self.device, non_blocking=self.is_cuda)
        # Nếu dataset trả về HU thô (int16), chuẩn hóa cửa sổ HU trực tiếp trên GPU
        if images.dtype in (torch.int16, torch.int32, torch.int64):
            images = window_hu_torch(images)
        return images, masks

    def run_epoch(self, loader, is_train=True, epoch=None, total_epochs=None):
        self.model.train(is_train)
        total_loss = 0.0
        total_samples = 0
        inter_sum = 0.0
        pred_sum = 0.0
        target_sum = 0.0

        phase = "Train" if is_train else "Val  "
        ep_prefix = f"Epoch [{epoch:02d}/{total_epochs:02d}] {phase}" if (epoch and total_epochs) else phase

        use_bar = self.show_progress and (tqdm is not None) and sys.stderr.isatty()
        iterator = tqdm(loader, desc=ep_prefix, leave=False, dynamic_ncols=True) if use_bar else loader

        grad_ctx = torch.enable_grad() if is_train else torch.no_grad()
        with grad_ctx:
            for batch in iterator:
                images, masks = self._prepare_batch(batch)
                bs = images.size(0)

                if is_train:
                    self.optimizer.zero_grad(set_to_none=True)

                with _autocast_context(self.use_amp):
                    logits = self.model(images)
                    loss = self.criterion(logits, masks)

                if is_train:
                    self.scaler.scale(loss).backward()
                    if self.grad_clip is not None and self.grad_clip > 0:
                        self.scaler.unscale_(self.optimizer)
                        nn.utils.clip_grad_norm_(self.model.parameters(), self.grad_clip)
                    self.scaler.step(self.optimizer)
                    self.scaler.update()

                total_loss += float(loss.item()) * bs
                total_samples += bs

                m = compute_segmentation_metrics(logits.detach(), masks)
                inter_sum += m["intersection"]
                pred_sum += m["pred_sum"]
                target_sum += m["target_sum"]

                if use_bar:
                    running_loss = total_loss / max(1, total_samples)
                    running_dice = (2.0 * inter_sum + 1e-6) / (pred_sum + target_sum + 1e-6)
                    iterator.set_postfix(loss=f"{running_loss:.4f}", dice=f"{running_dice:.4f}")

        eps = 1e-6
        epoch_loss = total_loss / max(1, total_samples)
        epoch_dice = (2.0 * inter_sum + eps) / (pred_sum + target_sum + eps)
        union_sum = pred_sum + target_sum - inter_sum
        epoch_iou = (inter_sum + eps) / (union_sum + eps)

        return {"loss": epoch_loss, "dice": epoch_dice, "iou": epoch_iou}

    def save_checkpoint(self, filename, epoch, val_metrics):
        path = self.save_dir / filename
        ckpt = {
            "epoch": int(epoch),
            "model_state_dict": self._unwrap_model().state_dict(),
            "optimizer_state_dict": self.optimizer.state_dict(),
            "scheduler_state_dict": self.scheduler.state_dict() if self.scheduler is not None else None,
            "scaler_state_dict": self.scaler.state_dict() if self.use_amp else None,
            "best_val_dice": float(self.best_val_dice),
            "no_improve": int(self.no_improve),
            "val_metrics": val_metrics,
            "history": self.history,
        }
        _atomic_torch_save(ckpt, path)
        return path

    def load_checkpoint(self, checkpoint_path, resume_training=True):
        """
        Khôi phục trạng thái từ file checkpoint (.pth).
        - resume_training=True : Khôi phục toàn bộ (model, optimizer, scheduler, scaler, epoch,
                                 best_val_dice, no_improve, history) để train tiếp sau khi ngắt kết nối.
        - resume_training=False: Chỉ nạp trọng số model_state_dict (dùng khi fine-tune / init_weights).
        """
        checkpoint_path = Path(checkpoint_path)
        if not checkpoint_path.exists():
            raise FileNotFoundError(f"Không tìm thấy file checkpoint: {checkpoint_path}")

        ckpt = torch.load(checkpoint_path, map_location=self.device)
        if not isinstance(ckpt, dict) or "model_state_dict" not in ckpt:
            self._unwrap_model().load_state_dict(ckpt)
            print(f"📥 Đã nạp trọng số mô hình (raw state_dict) từ: {checkpoint_path}")
            return 1

        self._unwrap_model().load_state_dict(ckpt["model_state_dict"])
        if not resume_training:
            print(f"📥 Đã nạp trọng số khởi tạo (init_weights) từ: {checkpoint_path}")
            return 1

        if "optimizer_state_dict" in ckpt and ckpt["optimizer_state_dict"] is not None:
            self.optimizer.load_state_dict(ckpt["optimizer_state_dict"])

        if self.scheduler is not None and ckpt.get("scheduler_state_dict") is not None:
            try:
                self.scheduler.load_state_dict(ckpt["scheduler_state_dict"])
            except Exception as e:
                print(f"⚠️ Không nạp được scheduler_state_dict ({e}), giữ cấu hình scheduler hiện tại.")

        if self.use_amp and ckpt.get("scaler_state_dict") is not None:
            try:
                self.scaler.load_state_dict(ckpt["scaler_state_dict"])
            except Exception as e:
                print(f"⚠️ Không nạp được scaler_state_dict ({e}), khởi tạo scaler mới.")

        last_epoch = int(ckpt.get("epoch", 0))
        self.start_epoch = last_epoch + 1
        self.best_val_dice = float(ckpt.get("best_val_dice", -1.0))
        self.no_improve = int(ckpt.get("no_improve", 0))

        # Khôi phục lịch sử huấn luyện từ checkpoint hoặc từ history.csv trên đĩa
        if isinstance(ckpt.get("history"), list) and len(ckpt["history"]) > 0:
            self.history = [r for r in ckpt["history"] if int(r.get("epoch", 0)) <= last_epoch]
        elif (self.save_dir / "history.csv").exists():
            try:
                df_hist = pd.read_csv(self.save_dir / "history.csv")
                df_hist = df_hist[df_hist["epoch"] <= last_epoch]
                self.history = df_hist.to_dict(orient="records")
            except Exception:
                self.history = []

        print(
            f"🔄 RESUME THÀNH CÔNG từ '{checkpoint_path}' | "
            f"Đã xong epoch {last_epoch} -> Tiếp tục từ epoch {self.start_epoch} | "
            f"Best Val Dice hiện tại: {self.best_val_dice:.4f}"
        )
        return self.start_epoch

    def fit(
        self,
        train_loader,
        val_loader=None,
        epochs=DEFAULT_EPOCHS,
        early_stopping_patience=0,
        resume_from=None,
    ):
        """
        Vòng lặp huấn luyện chính.
        Args:
            resume_from: None, đường dẫn tới file .pth, hoặc 'auto' (tự tìm <save_dir>/last_model.pth).
        """
        if resume_from:
            if str(resume_from).lower() == "auto":
                candidate = self.save_dir / "last_model.pth"
                if candidate.exists():
                    self.load_checkpoint(candidate, resume_training=True)
                else:
                    print(f"ℹ️  Chế độ --resume auto: Chưa có '{candidate}', bắt đầu huấn luyện mới từ Epoch 1.")
            else:
                self.load_checkpoint(resume_from, resume_training=True)

        # Nếu dùng CosineAnnealingLR và người dùng tăng tổng số epochs khi resume, đồng bộ lại T_max & LR
        if self.scheduler is not None and isinstance(self.scheduler, torch.optim.lr_scheduler.CosineAnnealingLR):
            if getattr(self.scheduler, "T_max", epochs) != epochs and epochs > 0:
                self.scheduler.T_max = max(1, epochs)
                if hasattr(self.scheduler, "_get_closed_form_lr"):
                    for group, lr_val in zip(self.optimizer.param_groups, self.scheduler._get_closed_form_lr()):
                        group["lr"] = lr_val
                    self.scheduler._last_lr = [group["lr"] for group in self.optimizer.param_groups]

        if self.start_epoch > epochs:
            print(
                f"✅ Mô hình đã hoàn thành {self.start_epoch - 1}/{epochs} epochs trong checkpoint trước đó. "
                f"(Hãy tăng --epochs lớn hơn {self.start_epoch - 1} nếu muốn huấn luyện thêm)."
            )
            return self.history

        n_batches_tr = len(train_loader)
        n_batches_va = len(val_loader) if val_loader is not None else 0
        print("=" * 95)
        print(
            f"🚀 BẮT ĐẦU HUẤN LUYỆN | Thiết bị: {self.device} | AMP: {self.use_amp} | "
            f"Epochs: {self.start_epoch} -> {epochs} | Batch: {n_batches_tr} train, {n_batches_va} val"
        )
        print("=" * 95)

        epoch_durations = []
        fit_start_time = time.time()

        for epoch in range(self.start_epoch, epochs + 1):
            ep_t0 = time.time()

            # 1. Pha Train
            tr_t0 = time.time()
            train_m = self.run_epoch(train_loader, is_train=True, epoch=epoch, total_epochs=epochs)
            train_time = time.time() - tr_t0

            # 2. Pha Validation
            va_t0 = time.time()
            if val_loader is not None:
                val_m = self.run_epoch(val_loader, is_train=False, epoch=epoch, total_epochs=epochs)
            else:
                val_m = train_m
            val_time = time.time() - va_t0

            # 3. Cập nhật Learning Rate Scheduler
            if self.scheduler is not None:
                if isinstance(self.scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
                    self.scheduler.step(val_m["dice"])
                else:
                    self.scheduler.step()

            lr = self.optimizer.param_groups[0]["lr"]
            elapsed = time.time() - ep_t0
            epoch_durations.append(elapsed)

            # Ước lượng thời gian còn lại (ETA) dựa trên trung bình trượt 5 epoch gần nhất
            recent_avg_sec = float(np.mean(epoch_durations[-5:]))
            remaining_epochs = epochs - epoch
            eta_sec = recent_avg_sec * remaining_epochs

            row = {
                "epoch": int(epoch),
                "lr": float(lr),
                "train_loss": round(train_m["loss"], 5),
                "train_dice": round(train_m["dice"], 5),
                "train_iou": round(train_m["iou"], 5),
                "val_loss": round(val_m["loss"], 5),
                "val_dice": round(val_m["dice"], 5),
                "val_iou": round(val_m["iou"], 5),
                "train_time_sec": round(train_time, 2),
                "val_time_sec": round(val_time, 2),
                "time_sec": round(elapsed, 2),
                "eta_sec": round(eta_sec, 2),
            }
            # Ghi đè nếu cùng epoch (an toàn khi resume lại), sau đó lưu history.csv
            self.history = [r for r in self.history if int(r.get("epoch", -1)) != epoch] + [row]
            _atomic_csv_save(pd.DataFrame(self.history), self.save_dir / "history.csv")

            is_best = val_m["dice"] > self.best_val_dice
            if is_best:
                self.best_val_dice = float(val_m["dice"])
                self.no_improve = 0
                self.save_checkpoint("best_model.pth", epoch, val_m)
            else:
                self.no_improve += 1

            # Luôn lưu last_model.pth sau khi đã cập nhật best_val_dice & no_improve để Resume chính xác
            self.save_checkpoint("last_model.pth", epoch, val_m)

            badge = "⭐ [BEST]" if is_best else ""
            print(
                f"Epoch [{epoch:02d}/{epochs:02d}] | LR: {lr:.2e} | "
                f"Train [Loss: {train_m['loss']:.4f}, Dice: {train_m['dice']:.4f}, IoU: {train_m['iou']:.4f}] | "
                f"Val [Loss: {val_m['loss']:.4f}, Dice: {val_m['dice']:.4f}, IoU: {val_m['iou']:.4f}] | "
                f"Time: {format_duration(elapsed)} (Tr:{train_time:.1f}s/Va:{val_time:.1f}s) | "
                f"ETA: {format_duration(eta_sec)} {badge}"
            )

            if early_stopping_patience > 0 and self.no_improve >= early_stopping_patience:
                print(
                    f"⏹️ Dừng sớm (Early Stopping) tại epoch {epoch} do Val Dice không cải thiện "
                    f"sau {early_stopping_patience} epochs liên tiếp."
                )
                break

        total_fit_time = time.time() - fit_start_time
        print("=" * 95)
        print(
            f"🎉 Hoàn tất! Best Val Dice: {self.best_val_dice:.4f} | "
            f"Tổng thời gian chạy: {format_duration(total_fit_time)} | Lưu tại: {self.save_dir}"
        )
        return self.history

def _build_criterion(loss_name):
    if loss_name == "tversky":
        return TverskyFocalLoss()
    return DiceBCELoss(bce_weight=0.5, dice_weight=0.5)


def _cmd_train(a):
    seed_everything(a.seed)
    train_loader, val_loader = create_dataloaders(
        cache_dir=a.cache_dir,
        batch_size=a.batch_size,
        num_workers=a.workers,
        in_channels=3,
        apply_lung=not a.no_lung_mask,
        window_on_cpu=not a.gpu_window,
        augment_train=not a.no_augment,
    )
    model = DuneUNet(in_channels=3, classes=1, features=a.features)
    criterion = _build_criterion(a.loss)
    optimizer = torch.optim.AdamW(model.parameters(), lr=a.lr, weight_decay=a.weight_decay)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=a.epochs, eta_min=1e-6)

    trainer = Trainer(
        model=model,
        criterion=criterion,
        optimizer=optimizer,
        scheduler=scheduler,
        use_amp=not a.no_amp,
        save_dir=a.save_dir,
        show_progress=not a.no_progress,
    )
    trainer.fit(
        train_loader,
        val_loader,
        epochs=a.epochs,
        early_stopping_patience=a.patience,
        resume_from=a.resume,
    )


def _cmd_overfit(a):
    """
    Chế độ Sanity Check: Lấy đúng N lát cắt CÓ KHỐI U từ cache_1mm và lặp lại việc học trên đó
    để kiểm chứng mô hình có khả năng hội tụ (Loss giảm sâu, Dice tiến sát 1.0) hay không.
    """
    seed_everything(a.seed)
    ds = LIDC25DDataset(
        cache_dir=a.cache_dir,
        split="train",
        in_channels=3,
        apply_lung=True,
        window_on_cpu=not a.gpu_window,
        augment=False,
    )
    pos_indices = ds.df.index[ds.df["has_tumor"] == 1].tolist()
    chosen = pos_indices[: a.num_samples] if len(pos_indices) >= a.num_samples else list(range(min(len(ds), a.num_samples)))
    subset = Subset(ds, chosen)
    loader = DataLoader(subset, batch_size=min(a.batch_size, len(chosen)), shuffle=True, num_workers=0)

    print(f"🔬 Chạy Overfit Sanity Check trên {len(chosen)} lát cắt có u...")
    model = DuneUNet(in_channels=3, classes=1, features=a.features)
    criterion = DiceBCELoss(bce_weight=0.5, dice_weight=0.5)
    optimizer = torch.optim.AdamW(model.parameters(), lr=a.lr, weight_decay=1e-5)

    trainer = Trainer(
        model=model,
        criterion=criterion,
        optimizer=optimizer,
        scheduler=None,
        use_amp=not a.no_amp,
        save_dir=a.save_dir,
        show_progress=not a.no_progress,
    )
    trainer.fit(loader, val_loader=loader, epochs=a.epochs, resume_from=a.resume)

import matplotlib.pyplot as plt
from matplotlib.widgets import Slider, CheckButtons
import numpy as np
import sys
from pathlib import Path

# Nạp các hàm từ src.medical_prep
sys.path.append(str(Path(__file__).resolve().parent))
from src.medical_prep import (load_nrrd, resize_and_pad_3d, clip_and_window_hu,
                              extract_lungs_mask, lung_keep_mask, apply_lung_mask)

# Thứ tự đúng với pipeline của tác giả: resize + pad trên HU thô, rồi mới kẹp cửa sổ.
STAGE_RAW    = "1. Ảnh Thô Nguyên Bản"
STAGE_RESIZE = "2. Resize 1mm + Pad (HU)"
STAGE_HU     = "3. Kẹp Cửa Sổ (HU)"
STAGE_LUNG   = "4. Tách Phổi + Cửa Sổ (chậm)"
STAGES = [STAGE_RAW, STAGE_RESIZE, STAGE_HU, STAGE_LUNG]
MASK_LABEL = "Hiện mask nốt"


class LazyViewer:
    """Mới mở chỉ có ảnh gốc. Mỗi bước chỉ được tính khi bạn tích chọn, và được cache lại."""

    def __init__(self, image_path, mask_path=None, title="Tiền xử lý CT"):
        print("Đang nạp ảnh CT thô...")
        self.vol_raw, self.spacing = load_nrrd(image_path)
        self.mask_raw = None
        if mask_path and Path(mask_path).exists():
            self.mask_raw, _ = load_nrrd(mask_path)
        else:
            print("(Không có mask.nrrd, bỏ qua lớp mặt nạ)")

        self.title = title
        self.vols = {STAGE_RAW: self.vol_raw}       # cache ảnh theo giai đoạn
        self.masks = {STAGE_RAW: self.mask_raw}     # cache mask nốt theo giai đoạn
        self.state = {s: (s == STAGE_RAW) for s in STAGES}   # mới đầu chỉ ảnh gốc
        self.show_mask = False
        self.n0 = self.vol_raw.shape[0]
        self.idx = self.n0 // 2
        self.axes, self.displays = [], []

        self.fig = plt.figure(figsize=(18, 7))
        self.fig.canvas.manager.set_window_title(title)

        # Ô tích chọn giai đoạn (+ mask)
        labels = STAGES + ([MASK_LABEL] if self.mask_raw is not None else [])
        actives = [self.state[s] for s in STAGES] + ([False] if self.mask_raw is not None else [])
        ax_chk = self.fig.add_axes([0.01, 0.36, 0.19, 0.28])
        ax_chk.set_title("Chọn bước hiển thị", fontsize=10, fontweight="bold")
        self.check = CheckButtons(ax_chk, labels, actives)
        self.check.on_clicked(self.on_check)

        # Thanh trượt Z
        ax_sl = self.fig.add_axes([0.25, 0.04, 0.65, 0.03])
        self.slider = Slider(ax_sl, "Lát cắt Z", 0, self.n0 - 1, valinit=self.idx, valstep=1)
        self.slider.on_changed(self.on_slide)

        self.redraw()

    # ---------- tính toán lười (chỉ chạy khi được chọn) ----------
    def get_vol(self, stage):
        if stage in self.vols:
            return self.vols[stage]
        if stage == STAGE_RESIZE:
            # resize trên HU thô; ảnh được pad bằng -1000 HU (mặc định của resize_and_pad_3d)
            self.vols[stage] = resize_and_pad_3d(self.vol_raw, self.spacing)
        elif stage == STAGE_HU:
            self.vols[stage] = clip_and_window_hu(self.get_vol(STAGE_RESIZE))
        elif stage == STAGE_LUNG:
            # tách phổi trên HU thô ở lưới gốc, resize mask về 1 mm, che ảnh bằng -1000, rồi cửa sổ.
            # Không hợp mask nốt vào (đó chỉ dành cho huấn luyện, dùng ở đây là rò rỉ nhãn).
            lung = extract_lungs_mask(np.maximum(self.vol_raw, -1000))
            if lung is None:
                raise RuntimeError("không tìm thấy phổi")
            lung_r = resize_and_pad_3d(lung, self.spacing, is_mask=True)
            self.vols[stage] = clip_and_window_hu(apply_lung_mask(self.get_vol(STAGE_RESIZE), lung_r))
        return self.vols[stage]

    def get_mask(self, stage):
        if self.mask_raw is None:
            return None
        if stage in self.masks:
            return self.masks[stage]
        # mọi bước sau bước 1 dùng chung mask nốt đã resize (nearest)
        if STAGE_RESIZE not in self.masks:
            self.masks[STAGE_RESIZE] = resize_and_pad_3d(self.mask_raw, self.spacing, is_mask=True)
        self.masks[stage] = self.masks[STAGE_RESIZE]
        return self.masks[stage]

    # ---------- vẽ ----------
    @staticmethod
    def _slice(arr, idx, n0):
        j = int(round(idx / max(n0 - 1, 1) * (arr.shape[0] - 1)))   # ánh xạ nếu số lát khác nhau
        return arr[j]

    def _status(self, text):
        self.fig.suptitle(text, fontsize=13)
        self.fig.canvas.draw()
        plt.pause(0.01)

    def redraw(self):
        for ax in self.axes:
            ax.remove()
        self.axes, self.displays = [], []

        active = [s for s in STAGES if self.state[s]]
        n = len(active)
        left, right, gap = 0.22, 0.99, 0.01
        w = (right - left - gap * (n - 1)) / n

        for i, stage in enumerate(active):
            self._status(f"Đang xử lý: {stage} ...")
            try:
                vol = self.get_vol(stage)
                mask = self.get_mask(stage) if self.show_mask else None
            except Exception as e:
                print(f"❌ Lỗi ở bước '{stage}': {e}")
                self._deactivate(stage)
                return self.redraw()

            ax = self.fig.add_axes([left + i * (w + gap), 0.12, w, 0.76])
            img = ax.imshow(self._slice(vol, self.idx, self.n0), cmap="gray")
            m_disp = None
            if mask is not None:
                m = self._slice(mask, self.idx, self.n0)
                m_disp = ax.imshow(np.ma.masked_where(m == 0, m), cmap="Reds",
                                   alpha=0.4, vmin=0, vmax=1)
            z, h, wd = vol.shape
            ax.set_title(f"{stage}\n(Shape: {z}x{h}x{wd})", fontsize=10, fontweight="bold")
            ax.axis("off")
            self.axes.append(ax)
            self.displays.append((vol, mask, img, m_disp))

        self.update_frames()

    def update_frames(self):
        for vol, mask, img, m_disp in self.displays:
            s = self._slice(vol, self.idx, self.n0)
            img.set_data(s)
            img.set_clim(vmin=s.min(), vmax=s.max())
            if m_disp is not None:
                m = self._slice(mask, self.idx, self.n0)
                m_disp.set_data(np.ma.masked_where(m == 0, m))
        self.fig.suptitle(f"{self.title} - Z: {self.idx}/{self.n0 - 1}", fontsize=13)
        self.fig.canvas.draw_idle()

    # ---------- sự kiện ----------
    def _deactivate(self, stage):
        self.state[stage] = False
        i = STAGES.index(stage)
        self.check.eventson = False       # bỏ tích mà không kích hoạt callback
        self.check.set_active(i)
        self.check.eventson = True

    def on_check(self, label):
        if label == MASK_LABEL:
            self.show_mask = not self.show_mask
        else:
            self.state[label] = not self.state[label]
            if not any(self.state.values()):      # luôn giữ ít nhất 1 ảnh
                self.state[STAGE_RAW] = True
                self.check.eventson = False
                self.check.set_active(0)
                self.check.eventson = True
        self.redraw()

    def on_slide(self, val):
        self.idx = int(val)
        self.update_frames()

def make_loaders(
    cache_dir,
    batch_size,
    workers,
    image_size,
    apply_lung,
    window_on_cpu,
    augment,
    limit_train=0,
    limit_val=0,
    seed=DEFAULT_SEED,
    shuffle_seed=None,
):
    """
    Dựng train/val loader. Nếu không có mẫu val thì trả val_loader=None (Trainer tự xử lý).
    `seed` quyết định tập con khi dùng --limit-* (phải giữ nguyên khi resume);
    `shuffle_seed` quyết định thứ tự xáo trộn (đổi khi resume để không lặp lại đúng thứ tự epoch 1).
    """
    train_ds = LIDC25DDataset(
        cache_dir, split="train", in_channels=3, image_size=image_size,
        apply_lung=apply_lung, window_on_cpu=window_on_cpu, augment=augment,
    )
    try:
        val_ds = LIDC25DDataset(
            cache_dir, split="val", in_channels=3, image_size=image_size,
            apply_lung=apply_lung, window_on_cpu=window_on_cpu, augment=False,
        )
    except ValueError:
        print("⚠️  Không có mẫu nào thuộc tập val -> dùng chỉ số train thay cho val (chỉ nên xảy ra khi thử nghiệm).")
        val_ds = None

    rng = np.random.RandomState(seed)

    def _limit(ds, n):
        if ds is None or n <= 0 or n >= len(ds):
            return ds
        return Subset(ds, sorted(rng.choice(len(ds), size=n, replace=False).tolist()))

    train_ds, val_ds = _limit(train_ds, limit_train), _limit(val_ds, limit_val)

    pin = torch.cuda.is_available()
    gen = torch.Generator().manual_seed(seed if shuffle_seed is None else shuffle_seed)
    train_loader = DataLoader(
        train_ds, batch_size=batch_size, shuffle=True, num_workers=workers,
        pin_memory=pin, drop_last=len(train_ds) > batch_size, generator=gen,
        persistent_workers=workers > 0,
    )
    val_loader = None
    if val_ds is not None:
        val_loader = DataLoader(
            val_ds, batch_size=batch_size, shuffle=False, num_workers=workers,
            pin_memory=pin, persistent_workers=workers > 0,
        )
    return train_loader, val_loader


def make_scheduler(name, optimizer, epochs):
    if name == "cosine":
        return torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=max(1, epochs), eta_min=ETA_MIN)
    if name == "plateau":
        return torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode="max", factor=0.5, patience=3)
    return None


def load_init_weights(model, path):
    """Nạp trọng số model (không nạp optimizer/epoch) để fine-tune từ checkpoint cũ."""
    ckpt = load_checkpoint_file(path, "cpu")
    meta = (ckpt.get("meta") or {}) if isinstance(ckpt, dict) else {}
    expect = {"features": model.features, "backbone": model.backbone_name}
    for k, v in expect.items():
        if k in meta and meta[k] != v:
            raise ValueError(
                f"Checkpoint {path} được train với {k}={meta[k]} nhưng lệnh hiện tại dùng {k}={v}. "
                f"Hãy đặt lại --{k} {meta[k]} (hoặc bỏ --init-weights)."
            )
    state = ckpt["model_state_dict"] if isinstance(ckpt, dict) and "model_state_dict" in ckpt else ckpt
    try:
        model.load_state_dict(state)
    except RuntimeError as e:
        raise ValueError(
            f"Không nạp được trọng số từ {path}: kiến trúc khác (thường do sai --features hoặc --backbone). "
            f"{str(e).splitlines()[0]}"
        ) from e
    print(f"📥 Đã nạp trọng số khởi tạo từ: {path}")

def resolve_resume_path(resume, save_dir):
    """'auto' -> <save_dir>/last_model.pth nếu có (không có thì train mới); ngược lại là đường dẫn cụ thể."""
    if not resume:
        return None
    if str(resume).lower() == "auto":
        cand = Path(save_dir) / "last_model.pth"
        if cand.exists():
            return cand
        print(f"ℹ️  --resume auto: chưa có {cand} -> bắt đầu train mới.")
        return None
    path = Path(resume)
    if not path.exists():
        raise FileNotFoundError(f"Không tìm thấy checkpoint để resume: {path.resolve()}")
    return path


def restore_scheduler(trainer, ckpt, a):
    """
    Quyết định cách nối tiếp learning rate sau khi trainer.load_checkpoint đã nạp optimizer:
      - cùng loại scheduler, cosine, T_max khớp --epochs -> nạp lại trạng thái (tiếp tục đúng đường cũ);
      - cùng loại, plateau                                -> nạp lại trạng thái;
      - cosine nhưng --epochs đổi / khác loại scheduler   -> chu kỳ mới: lr đặt lại = --lr cho phần epoch còn lại.
    """
    cur = trainer.scheduler
    if cur is None:
        return
    done_epoch = int(ckpt["epoch"])
    saved_type = ckpt.get("scheduler_type")
    saved_state = ckpt.get("scheduler_state_dict")
    is_cosine = isinstance(cur, torch.optim.lr_scheduler.CosineAnnealingLR)

    if saved_type == type(cur).__name__ and saved_state is not None:
        if is_cosine and saved_state.get("T_max") != a.epochs:
            reason = f"--epochs đổi ({saved_state.get('T_max')} -> {a.epochs})"
        else:
            cur.load_state_dict(saved_state)
            print(f"📈 Tiếp tục scheduler {saved_type} đúng đường cũ | lr hiện tại = {trainer.optimizer.param_groups[0]['lr']:.2e}")
            return
    else:
        reason = f"scheduler đổi ({saved_type} -> {type(cur).__name__})"

    for g in trainer.optimizer.param_groups:
        base = a.lr * g.get("lr_mult", 1.0)
        g["lr"] = base
        g["initial_lr"] = base
    if is_cosine:
        remaining = max(1, a.epochs - done_epoch)
        trainer.scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
            trainer.optimizer, T_max=remaining, eta_min=ETA_MIN
        )
        print(f"🔄 {reason}: khởi động chu kỳ cosine mới cho {remaining} epoch còn lại, lr = {a.lr:.2e}")
    else:
        print(f"🔄 {reason}: dùng scheduler mới, lr đặt lại = {a.lr:.2e}")

def run_train(a):
    seed_everything(a.seed)
    if a.resume and a.init_weights:
        raise ValueError("Chỉ dùng MỘT trong hai: --resume (tiếp tục đầy đủ) hoặc --init-weights (chỉ nạp trọng số).")
    resume_path = resolve_resume_path(a.resume, a.save_dir)

    csv_path = ensure_index(a.cache_dir, rebuild=a.rebuild_index)
    summarize_index(csv_path)

    model = DuneUNet(
        in_channels=3, classes=1, features=a.features,
        backbone=a.backbone, pretrained=not a.no_pretrained,
    )
    if a.init_weights:
        load_init_weights(model, a.init_weights)
    n_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    use_backbone = model.has_backbone
    bb_mult = a.backbone_lr_mult if (use_backbone and not a.no_pretrained) else 1.0
    if use_backbone:
        n_bb = sum(p.numel() for p in model.encoder.backbone.parameters())
        print(f"🧬 Backbone {model.backbone_name}: {n_bb:,} tham số "
              f"({'pretrained' if not a.no_pretrained else 'khởi tạo từ đầu'}) | "
              f"lr backbone = lr x {bb_mult} | đóng băng: {a.freeze_backbone_epochs} epoch")

    criterion = _build_criterion(a.loss)
    optimizer = torch.optim.AdamW(model.param_groups(a.lr, bb_mult), lr=a.lr, weight_decay=a.weight_decay)
    scheduler = make_scheduler(a.scheduler, optimizer, a.epochs)

    trainer = Trainer(
        model=model,
        criterion=criterion,
        optimizer=optimizer,
        scheduler=scheduler,
        use_amp=not a.no_amp,
        save_dir=a.save_dir,
        grad_clip=a.grad_clip,
    )
    trainer.extra_meta = {"features": int(a.features), "in_channels": 3, "loss": a.loss,
                          "backbone": model.backbone_name}
    trainer.freeze_backbone_epochs = int(a.freeze_backbone_epochs)

    if resume_path is not None:
        print(f"♻️  Resume từ: {resume_path}")
        ckpt = trainer.load_checkpoint(resume_path, load_scheduler=False, check_meta={"features": int(a.features), "backbone": model.backbone_name})
        restore_scheduler(trainer, ckpt, a)
        wd = trainer.optimizer.param_groups[0].get("weight_decay")
        if wd is not None and abs(wd - a.weight_decay) > 1e-12:
            print(f"ℹ️  weight_decay lấy từ checkpoint ({wd}), bỏ qua --weight-decay {a.weight_decay}.")
        print(f"   Đã xong {trainer.start_epoch - 1} epoch | Best Val Dice = {trainer.best_val_dice:.4f}")

    train_loader, val_loader = make_loaders(
        cache_dir=a.cache_dir,
        batch_size=a.batch_size,
        workers=a.workers,
        image_size=a.image_size,
        apply_lung=not a.no_lung_mask,
        window_on_cpu=not a.gpu_window,
        augment=not a.no_augment,
        limit_train=a.limit_train,
        limit_val=a.limit_val,
        seed=a.seed,
        shuffle_seed=a.seed + trainer.start_epoch - 1,
    )
    n_tr = len(train_loader.dataset)
    n_va = len(val_loader.dataset) if val_loader is not None else 0
    print(f"📦 Số lát dùng thực tế: train={n_tr} | val={n_va}" + (" (đã giới hạn bởi --limit-*)" if (a.limit_train or a.limit_val) else ""))

    run_info = {
        "args": {k: (str(v) if isinstance(v, Path) else v) for k, v in vars(a).items() if k != "fn"},
        "device": str(trainer.device),
        "amp": trainer.use_amp,
        "num_params": n_params,
        "train_samples": n_tr,
        "val_samples": n_va,
        "resumed_from": str(resume_path) if resume_path is not None else None,
        "start_epoch": trainer.start_epoch,
        "started_at": time.strftime("%Y-%m-%d %H:%M:%S"),
    }
    Path(a.save_dir).mkdir(parents=True, exist_ok=True)
    cfg_name = "config.json" if resume_path is None else f"config_resume_ep{trainer.start_epoch - 1}.json"
    with open(Path(a.save_dir) / cfg_name, "w", encoding="utf-8") as f:
        json.dump(run_info, f, ensure_ascii=False, indent=2)

    print(
        f"🧠 DuneUNet(features={a.features}) | {n_params:,} tham số | "
        f"loss={a.loss} | lr={a.lr} | scheduler={a.scheduler} | batch={a.batch_size} | backbone={model.backbone_name}"
    )
    history = trainer.fit(
        train_loader, val_loader, epochs=a.epochs, early_stopping_patience=a.patience
    )
    print(f"💾 Checkpoint tốt nhất: {Path(a.save_dir) / 'best_model.pth'}")
    return history

def _expect_error(exc_type, fragment, fn):
    """Khẳng định fn() ném exc_type và thông báo chứa `fragment`."""
    try:
        fn()
    except exc_type as e:
        assert fragment in str(e), f"Thông báo lỗi thiếu '{fragment}': {e}"
        return
    raise AssertionError(f"Phải ném {exc_type.__name__} chứa '{fragment}'")


def _selftest_basic(tmp, cache):

    assert len(hist) == a.epochs
    assert hist[-1]["train_loss"] < hist[0]["train_loss"], "Train loss phải giảm"
    assert all(np.isfinite(r["train_loss"]) and np.isfinite(r["val_loss"]) for r in hist), "Loss có NaN/Inf"
    with open(save / "config.json", encoding="utf-8") as f:
        info = json.load(f)
    assert info["train_samples"] > 0 and info["val_samples"] > 0 and info["num_params"] > 0

    a2 = _default_args(
        cache_dir=str(cache), save_dir=str(tmp / "ckpt2"), epochs=2, scheduler="plateau",
        loss="tversky", init_weights=str(save / "best_model.pth"), limit_train=6,
    )
    hist2 = run_train(a2)
    assert len(hist2) == 2 and all(np.isfinite(r["train_loss"]) for r in hist2)

    a3 = _default_args(
        cache_dir=str(cache), save_dir=str(tmp / "ckpt3"), epochs=3, scheduler="none", patience=1,
    )
    hist3 = run_train(a3)
    assert 1 <= len(hist3) <= 3
    return hist


def _selftest_state_roundtrip(tmp, ckpt_path):
    """Nạp checkpoint vào Trainer MỚI (cố tình đặt lr=1.0) rồi so khớp từng thành phần với nội dung file."""
    ck = load_checkpoint_file(ckpt_path)
    model = DuneUNet(in_channels=3, classes=1, features=8)
    opt = torch.optim.AdamW(model.parameters(), lr=1.0)
    tr = Trainer(model, _build_criterion("dice_bce"), opt, make_scheduler("cosine", opt, 6),
                 use_amp=False, save_dir=tmp / "roundtrip")
    tr.load_checkpoint(ckpt_path, load_scheduler=True, check_meta={"features": 8})

    cur_w = tr.model.state_dict()
    for k, v in ck["model_state_dict"].items():
        assert torch.equal(cur_w[k].cpu(), v.cpu()), f"Trọng số {k} không khớp"
    saved_st, cur_st = ck["optimizer_state_dict"]["state"], tr.optimizer.state_dict()["state"]
    assert len(saved_st) > 0 and saved_st.keys() == cur_st.keys(), "Thiếu moment của optimizer"
    for i in saved_st:
        assert torch.equal(saved_st[i]["exp_avg"], cur_st[i]["exp_avg"]), "exp_avg không khớp"
        assert torch.equal(saved_st[i]["exp_avg_sq"], cur_st[i]["exp_avg_sq"]), "exp_avg_sq không khớp"
        assert float(saved_st[i]["step"]) == float(cur_st[i]["step"]), "step không khớp"
    assert tr.optimizer.param_groups[0]["lr"] == ck["optimizer_state_dict"]["param_groups"][0]["lr"] != 1.0
    assert tr.scheduler.last_epoch == ck["scheduler_state_dict"]["last_epoch"] == ck["epoch"]
    assert tr.best_val_dice == ck["best_val_dice"] and tr.no_improve == ck["no_improve"]
    assert tr.start_epoch == ck["epoch"] + 1 and len(tr.history) == ck["epoch"]


def _selftest_resume(tmp, cache, reference_hist):
    """Kiểm thử resume: ngắt giữa chừng thật (KeyboardInterrupt) -> tiếp tục -> kéo dài -> các tình huống lỗi."""
    print("\n" + "#" * 65 + "\n# KIỂM THỬ RESUME\n" + "#" * 65)
    ref_lr = [r["lr"] for r in reference_hist]
    save = tmp / "resume_run"
    common = dict(cache_dir=str(cache), epochs=6)

    # --- 1) Ngắt ở epoch 4 (giả lập Ctrl+C), kiểm tra checkpoint dừng đúng ở epoch 3 ---
    orig = Trainer.run_epoch
    calls = {"n": 0}

    def flaky(self, loader, is_train=True):
        if is_train:
            calls["n"] += 1
            if calls["n"] == 4:
                raise KeyboardInterrupt
        return orig(self, loader, is_train)

    Trainer.run_epoch = flaky
    try:
        h_int = run_train(_default_args(save_dir=str(save), **common))
    finally:
        Trainer.run_epoch = orig
    assert [r["epoch"] for r in h_int] == [1, 2, 3], "Ngắt ở epoch 4 phải còn đúng 3 epoch"
    ck3 = load_checkpoint_file(save / "last_model.pth")
    assert ck3["epoch"] == 3 and ck3["scheduler_state_dict"] is not None and ck3["meta"]["features"] == 8
    assert not list(save.glob("*.tmp")), "Không được để lại file tạm"
    shutil.copy(save / "last_model.pth", save / "ep3.pth")
    loss_ep3 = h_int[-1]["train_loss"]

    # --- 2) Resume với đúng --epochs: phải chạy epoch 4..6, lr khớp lượt chạy liền mạch ---
    h_res = run_train(_default_args(save_dir=str(save), resume=str(save / "last_model.pth"), **common))
    assert [r["epoch"] for r in h_res] == [1, 2, 3, 4, 5, 6], f"History phải liền mạch: {[r['epoch'] for r in h_res]}"
    assert np.allclose([r["lr"] for r in h_res], ref_lr, rtol=1e-6, atol=1e-12), \
        f"Đường lr sau resume phải trùng lượt chạy liền mạch:\n{[r['lr'] for r in h_res]}\n{ref_lr}"
    assert h_res[3]["train_loss"] < h_res[0]["train_loss"], "Epoch 4 sau resume không được quay lại mức khởi tạo"
    assert abs(h_res[3]["train_loss"] - loss_ep3) < 0.15, "Loss sau resume phải liên tục với trước khi ngắt"
    csv_hist = pd.read_csv(save / "history.csv")
    assert list(csv_hist["epoch"]) == [1, 2, 3, 4, 5, 6]
    ck6 = load_checkpoint_file(save / "last_model.pth")
    assert ck6["epoch"] == 6
    assert abs(ck6["best_val_dice"] - csv_hist["val_dice"].max()) < 1e-4, "best_val_dice phải nhất quán với history"
    assert (save / "config_resume_ep3.json").exists() and (save / "config.json").exists()
    print("✔ ngắt giữa chừng + resume đúng đường lr: OK")

    _selftest_state_roundtrip(tmp, save / "ep3.pth")
    print("✔ khôi phục đủ trạng thái (trọng số, moment optimizer, scheduler, best, bộ đếm): OK")

    # --- 3) Đã đủ epoch -> không train gì, không ghi đè ---
    mtime = (save / "last_model.pth").stat().st_mtime_ns
    h_done = run_train(_default_args(save_dir=str(save), resume=str(save / "last_model.pth"), **common))
    assert len(h_done) == 6 and (save / "last_model.pth").stat().st_mtime_ns == mtime, "Không được ghi đè khi đã đủ epoch"
    print("✔ resume khi đã đủ epoch: OK")

    # --- 4) Kéo dài 6 -> 9 epoch sang thư mục MỚI: chu kỳ cosine mới, history được mang theo ---
    save_ext = tmp / "resume_ext"
    h_ext = run_train(_default_args(save_dir=str(save_ext), resume=str(save / "last_model.pth"),
                                    cache_dir=str(cache), epochs=9))
    assert [r["epoch"] for r in h_ext] == list(range(1, 10))
    assert h_ext[6]["lr"] > 1e-3, f"Kéo dài phải khởi động lại lr, nhận được {h_ext[6]['lr']:.2e}"
    assert h_ext[-1]["lr"] < h_ext[6]["lr"], "lr phải giảm dần trong chu kỳ mới"
    assert [r["train_loss"] for r in h_ext[:6]] == [r["train_loss"] for r in h_res], "6 epoch đầu phải được mang sang nguyên vẹn"
    assert (save / "best_model.pth").stat().st_mtime_ns < (save_ext / "last_model.pth").stat().st_mtime_ns
    print("✔ kéo dài epoch (6 -> 9) sang thư mục mới: OK")

    # --- 5) Resume từ checkpoint CŨ trong cùng thư mục: history bị cắt, bản gốc được sao lưu ---
    h_old = run_train(_default_args(save_dir=str(save), resume=str(save / "ep3.pth"), **common))
    assert [r["epoch"] for r in h_old] == [1, 2, 3, 4, 5, 6]
    backup = pd.read_csv(save / "history_before_resume.csv")
    assert len(backup) == 6, "Phải sao lưu history trước khi cắt"
    print("✔ resume từ checkpoint cũ (cắt history + sao lưu): OK")

    # --- 6) Đổi loại scheduler khi resume -> vẫn chạy, lr đặt lại ---
    h_sw = run_train(_default_args(save_dir=str(tmp / "resume_sw"), resume=str(save / "ep3.pth"),
                                   scheduler="plateau", **common))
    assert [r["epoch"] for r in h_sw] == [1, 2, 3, 4, 5, 6]
    print("✔ đổi scheduler khi resume: OK")

    # --- 7) --resume auto ---
    auto_dir = tmp / "resume_auto"
    h_a1 = run_train(_default_args(save_dir=str(auto_dir), resume="auto", cache_dir=str(cache), epochs=2))
    assert [r["epoch"] for r in h_a1] == [1, 2], "auto khi chưa có checkpoint phải train mới"
    h_a2 = run_train(_default_args(save_dir=str(auto_dir), resume="auto", cache_dir=str(cache), epochs=3))
    assert [r["epoch"] for r in h_a2] == [1, 2, 3], "auto khi đã có checkpoint phải nối tiếp"
    print("✔ --resume auto: OK")

    # --- 8) Các tình huống lỗi phải có thông báo rõ ràng ---
    _expect_error(ValueError, "features", lambda: run_train(
        _default_args(save_dir=str(tmp / "err1"), resume=str(save / "last_model.pth"), features=16, **common)))
    _expect_error(ValueError, "MỘT trong hai", lambda: run_train(
        _default_args(save_dir=str(tmp / "err2"), resume=str(save / "last_model.pth"),
                      init_weights=str(save / "best_model.pth"), **common)))
    _expect_error(FileNotFoundError, "resume", lambda: run_train(
        _default_args(save_dir=str(tmp / "err3"), resume=str(tmp / "khong_co.pth"), **common)))
    assert not (tmp / "err1").exists() or not list((tmp / "err1").glob("*.pth")), "Lỗi tương thích không được để lại checkpoint"
    print("✔ các tình huống lỗi: OK")


def _bb_hash(bb):
    """Chữ ký của toàn bộ trọng số + thống kê BatchNorm của backbone (đổi khi có bất kỳ cập nhật nào)."""
    return sum(v.double().abs().sum().item() for v in bb.state_dict().values() if v.is_floating_point())


def _run_recorded(args, interrupt_at=None):
    """Chạy run_train, ghi lại trạng thái backbone ở từng epoch train; có thể giả lập Ctrl+C ở epoch thứ N."""
    rec, calls, orig = [], {"n": 0}, Trainer.run_epoch

    def wrapped(self, loader, is_train=True):
        if not is_train:
            return orig(self, loader, is_train)
        calls["n"] += 1
        if interrupt_at and calls["n"] == interrupt_at:
            raise KeyboardInterrupt
        core = self._unwrap_model()
        bb = core.encoder.backbone if core.has_backbone else None
        h0 = _bb_hash(bb) if bb is not None else None
        out = orig(self, loader, is_train)
        if bb is not None:
            rec.append(dict(frozen=core.encoder_frozen, bb_train=bb.training,
                            rg=any(q.requires_grad for q in bb.parameters()), changed=_bb_hash(bb) != h0))
        return out

    Trainer.run_epoch = wrapped
    try:
        hist = run_train(args)
    finally:
        Trainer.run_epoch = orig
    return hist, rec


def _selftest_backbone(tmp, cache):
    """Chuyển đổi có/không backbone: lịch đóng băng, lr theo nhóm, resume qua mốc mở khóa, tương thích checkpoint."""
    print("\n" + "#" * 65 + "\n# KIỂM THỬ BACKBONE (mambavision_mock)\n" + "#" * 65)
    col = lambda rec, k: [r[k] for r in rec]
    common = dict(cache_dir=str(cache), backbone="mambavision_mock", features=8, backbone_lr_mult=0.1,
                  freeze_backbone_epochs=2, epochs=4)

    # --- 1) Lịch đóng băng 2 epoch đầu rồi mở: trạng thái, chế độ BN và cập nhật trọng số đúng từng epoch ---
    save = tmp / "bb_run"
    hist, rec = _run_recorded(_default_args(save_dir=str(save), **common))
    assert col(rec, "frozen") == [True, True, False, False], col(rec, "frozen")
    assert col(rec, "rg") == [False, False, True, True], "requires_grad của backbone sai lịch"
    assert col(rec, "bb_train") == [False, False, True, True], "BN của backbone phải ở eval khi đóng băng"
    assert col(rec, "changed") == [False, False, True, True], "Backbone đóng băng không được thay đổi dù chỉ một giá trị"
    assert all(np.isfinite(r["train_loss"]) for r in hist) and hist[-1]["train_loss"] < hist[0]["train_loss"]
    ck = load_checkpoint_file(save / "last_model.pth")
    assert ck["meta"]["backbone"] == "mambavision_mock"
    g = ck["optimizer_state_dict"]["param_groups"]
    # (cuối chu kỳ cosine cả hai nhóm đều về eta_min tuyệt đối nên so tỉ lệ ở lr gốc `initial_lr`)
    assert len(g) == 2 and g[0]["lr_mult"] == 1.0 and g[1]["lr_mult"] == 0.1
    assert abs(g[1]["initial_lr"] / g[0]["initial_lr"] - 0.1) < 1e-9, f"lr gốc theo nhóm sai: {[x['initial_lr'] for x in g]}"
    print("✔ lịch đóng băng 2 epoch đầu + lr backbone x0.1: OK")

    # --- 2) Resume ngay SAU mốc mở khóa: epoch 3,4 phải ở trạng thái mở ---
    s2 = tmp / "bb_resume_a"
    _run_recorded(_default_args(save_dir=str(s2), **common), interrupt_at=3)
    h2, rec2 = _run_recorded(_default_args(save_dir=str(s2), resume=str(s2 / "last_model.pth"), **common))
    assert [r["epoch"] for r in h2] == [1, 2, 3, 4]
    assert col(rec2, "frozen") == [False, False] and col(rec2, "changed") == [True, True], "Resume sau mốc: backbone phải mở"
    # --- 3) Resume TRƯỚC mốc (giữa giai đoạn đóng băng): epoch 2 phải đóng băng lại ---
    s3 = tmp / "bb_resume_b"
    _run_recorded(_default_args(save_dir=str(s3), **common), interrupt_at=2)
    h3, rec3 = _run_recorded(_default_args(save_dir=str(s3), resume=str(s3 / "last_model.pth"), **common))
    assert [r["epoch"] for r in h3] == [1, 2, 3, 4]
    assert col(rec3, "frozen") == [True, False, False] and col(rec3, "changed") == [False, True, True], \
        f"Resume giữa giai đoạn đóng băng sai: {rec3}"
    print("✔ resume trước/sau mốc mở khóa (trạng thái suy ra từ số epoch): OK")

    # --- 4) Kéo dài epoch khi có 2 nhóm lr: tỉ lệ lr giữa các nhóm được giữ ---
    s4 = tmp / "bb_extend"
    h4 = run_train(_default_args(save_dir=str(s4), resume=str(save / "last_model.pth"), **{**common, "epochs": 6}))
    g4 = load_checkpoint_file(s4 / "last_model.pth")["optimizer_state_dict"]["param_groups"]
    assert g4[0]["initial_lr"] == 3e-3 and abs(g4[1]["initial_lr"] / g4[0]["initial_lr"] - 0.1) < 1e-9, \
        [x["initial_lr"] for x in g4]
    assert h4[4]["lr"] > 1e-3, f"Chu kỳ mới phải khởi động lại lr, nhận {h4[4]['lr']:.2e}"
    print("✔ kéo dài epoch giữ tỉ lệ lr giữa decoder/backbone: OK")

    # --- 5) Đóng băng suốt quá trình (-1): backbone bất động, decoder vẫn học ---
    h5, rec5 = _run_recorded(_default_args(save_dir=str(tmp / "bb_frozen"), **{**common, "freeze_backbone_epochs": -1}))
    assert all(col(rec5, "frozen")) and not any(col(rec5, "changed")) and h5[-1]["train_loss"] < h5[0]["train_loss"]
    print("✔ đóng băng suốt quá trình (-1): OK")

    # --- 6) Cờ đóng băng không làm hỏng baseline ---
    h6, rec6 = _run_recorded(_default_args(save_dir=str(tmp / "base_freezeflag"), cache_dir=str(cache),
                                           freeze_backbone_epochs=2, epochs=2))
    assert len(h6) == 2 and rec6 == []
    print("✔ baseline bỏ qua cờ đóng băng: OK")

    # --- 7) Chuyển cấu hình sai phải báo lỗi rõ, không âm thầm nạp bậy ---
    base_ck = tmp / "ckpt" / "last_model.pth"       # checkpoint baseline từ phần kiểm thử cơ bản
    _expect_error(ValueError, "backbone", lambda: run_train(_default_args(
        save_dir=str(tmp / "e1"), cache_dir=str(cache), resume=str(save / "last_model.pth"), epochs=8)))   # mock -> baseline
    _expect_error(ValueError, "backbone", lambda: run_train(_default_args(
        save_dir=str(tmp / "e2"), cache_dir=str(cache), resume=str(base_ck), backbone="mambavision_mock", epochs=8)))  # baseline -> mock
    _expect_error(ValueError, "backbone", lambda: run_train(_default_args(
        save_dir=str(tmp / "e3"), cache_dir=str(cache), init_weights=str(base_ck), backbone="mambavision_mock")))
    # checkpoint định dạng cũ (meta chưa có 'backbone') vẫn resume được ở chế độ baseline
    old = load_checkpoint_file(base_ck); old["meta"].pop("backbone", None)
    old_dir = tmp / "old_fmt"; old_dir.mkdir(); torch.save(old, old_dir / "last_model.pth")
    h7 = run_train(_default_args(save_dir=str(old_dir), cache_dir=str(cache), resume=str(old_dir / "last_model.pth"), epochs=8))
    assert h7[-1]["epoch"] == 8
    print("✔ chuyển cấu hình sai báo lỗi rõ + checkpoint định dạng cũ vẫn resume: OK")


def run_selftest(_):
    print("Đang chạy kiểm thử toàn trình train_main.py (dữ liệu giả lập)...")
    tmp = Path(tempfile.mkdtemp())
    try:
        cache = tmp / "cache"
        _make_synthetic_cache(cache)
        hist = _selftest_basic(tmp, cache)
        _selftest_resume(tmp, cache, hist)
        _selftest_backbone(tmp, cache)
        print(
            f"\n✅ selftest train_main.py: OK! "
            f"(run 1: train_loss {hist[0]['train_loss']:.4f} -> {hist[-1]['train_loss']:.4f}, "
            f"val_dice cuối = {hist[-1]['val_dice']:.4f})"
        )
    finally:
        shutil.rmtree(tmp, ignore_errors=True)

def build_parser():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = ap.add_subparsers(dest="cmd", required=True)

    s = sub.add_parser("selftest", help="Kiểm thử toàn trình bằng dữ liệu giả lập (vài chục giây trên CPU)")
    s.set_defaults(fn=run_selftest)

    t = sub.add_parser("train", help="Huấn luyện chính thức trên cache_1mm")
    t.add_argument("--cache-dir", default="data/cache_1mm")
    t.add_argument("--save-dir", default="checkpoints/dune_unet_25d")
    t.add_argument("--batch-size", type=int, default=DEFAULT_BATCH_SIZE)
    t.add_argument("--epochs", type=int, default=DEFAULT_EPOCHS, help="TỔNG số epoch đích (kể cả khi resume)")
    t.add_argument("--lr", type=float, default=DEFAULT_LR)
    t.add_argument("--weight-decay", type=float, default=1e-4)
    t.add_argument("--grad-clip", type=float, default=1.0, help="0 = tắt")
    t.add_argument("--features", type=int, default=64, help="Số filter cơ sở (32 nhẹ, 64 chuẩn)")
    t.add_argument("--image-size", type=int, default=DEFAULT_IMAGE_SIZE,
                   help="Kích thước sau pad/crop ở giữa (KHÔNG resize; giữ 512 khi train thật)")
    t.add_argument("--workers", type=int, default=2)
    t.add_argument("--patience", type=int, default=10, help="Early stopping patience (0 = tắt)")
    t.add_argument("--loss", choices=["dice_bce", "tversky"], default="dice_bce")
    t.add_argument("--scheduler", choices=["cosine", "plateau", "none"], default="cosine")
    t.add_argument("--backbone", default="none",
                   help="none | mambavision_t/t2/s/b/l/l2 | mambavision_mock (xem phần CHỌN BACKBONE ở đầu file)")
    t.add_argument("--no-pretrained", action="store_true", help="MambaVision: khởi tạo từ đầu, không nạp pretrained")
    t.add_argument("--freeze-backbone-epochs", type=int, default=0,
                   help="Đóng băng backbone N epoch đầu rồi mở khóa (-1 = đóng băng suốt, 0 = không đóng băng)")
    t.add_argument("--backbone-lr-mult", type=float, default=0.1,
                   help="Hệ số lr cho backbone pretrained so với phần còn lại")
    t.add_argument("--resume", default=None,
                   help="Đường dẫn checkpoint (.pth) để tiếp tục đầy đủ, hoặc 'auto' = last_model.pth trong --save-dir")
    t.add_argument("--init-weights", default=None, help="Chỉ nạp trọng số model từ checkpoint (không dùng chung với --resume)")
    t.add_argument("--limit-train", type=int, default=0, help="Chỉ dùng N lát train ngẫu nhiên (0 = tất cả)")
    t.add_argument("--limit-val", type=int, default=0, help="Chỉ dùng N lát val ngẫu nhiên (0 = tất cả)")
    t.add_argument("--rebuild-index", action="store_true", help="Dựng lại slices.csv trước khi train")

    t.add_argument("--no-lung-mask", action="store_true")
    t.add_argument("--no-augment", action="store_true")
    t.add_argument("--no-amp", action="store_true")
    t.add_argument("--seed", type=int, default=DEFAULT_SEED)
    t.set_defaults(fn=run_train)
